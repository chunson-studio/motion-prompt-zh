// 甩镜头转场 / Whip pan
// 分类: camera (镜头运动与转场) · element 19 · 子类别: 转场
// 示例表现: 相机快速横向甩动，在条带状运动模糊最强时换入下一个场景。
// 动画参数与实现规则:
//   - 衔接：1.7–2.5 s，持续 0.8 s；前 0.4 s，前一镜头甩出；后 0.4 s，后一镜头甩入并稳定。
//   - 前半段：前一镜头偏航角从 0→+0.45 rad，约 26°，使用二次缓入缓出；后半段：后一镜头偏航角从 -0.45 rad→0，使用相同缓动。
//   - 切换：在进度 50% 时直接替换场景，此时模糊最强，剪切被隐藏。
//   - 模糊：在 30 fps 下，对分布于前后 ±1.6 帧范围的 9 次渲染取平均。
//   - 相机：高度 1.4 m，除甩动期间外均固定；主体不动，A 为人物，B 为汽车和城市。
//   - 每 4.7 s 交替执行 A→B 和 B→A，完整往返为 9.4 s；右上角读数显示“A → B”或“B → A”。
// 适用场景: 适合在保持动感的同时切换地点或话题。
// 标签: whip pan, swish pan, motion blur, fast pan, transition, energy, 甩镜头, 转场, 运动模糊, 横摇, 速度, 衔接
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit camera -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#043
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 19
// 19 ホイップパン（ブレで次の場面へ）
CELLS[18] = shot(abCell((k, w, h, u, A, B) => {
  const t0 = 1.7, t1 = 2.5;
  const at = (kk, s) => {   // s: 0..1 の進み。前半 A を右へ振り、後半 B が左から止まる
    if (s < .5) world(kk, w, h, cam({ y: 1.4, yaw: eIOs(s * 2) * 0.45 }), A === sceneA ? { person: { x: -0.6, z: 7 } } : SB);
    else world(kk, w, h, cam({ y: 1.4, yaw: -(1 - eIOs((s - .5) * 2)) * 0.45 }), B === sceneA ? { person: { x: -0.6, z: 7 } } : SB);
  };
  if (u < t0) return A(k, w, h);
  if (u > t1) return B(k, w, h);
  const s = P(u, t0, t1), ds = 1 / FPS / (t1 - t0);
  mblur(k, at, clamp(s - ds * 1.6), clamp(s + ds * 1.6), 9);
}));
