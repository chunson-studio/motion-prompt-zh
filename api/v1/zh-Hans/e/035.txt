// FPV 穿越机镜头 / FPV drone shot
// 分类: camera (镜头运动与转场) · element 11 · 子类别: 拍摄方式与时间效果
// 示例表现: 以第一人称视角低空高速飞过两排树木之间，并左右倾斜穿行。
// 动画参数与实现规则:
//   - 相机：以恒定 14 m/s 向前飞行，标示为 50km/h，不使用缓动；俯仰角固定为 -0.05 rad。
//   - 穿行摆动：T = 2π×u/4.7，每 4.7 s 摆动两次；横向位置为 0.6·sin(2T-0.4) m，高度为 1.4 + 0.5·sin(2T) m。
//   - 侧倾：横滚角为 0.22·sin(2T+0.6) rad，约 ±12.6°；偏航角为 0.05·sin(2T) rad，约 ±2.9°。
//   - 没有主体。两侧各距中心 3.6 m 的位置种植高 4.2 m 的树，每隔 8 m 一棵，两排前后错开 4 m；树木快速迎面掠过。
//   - 速度线：10 条半透明灰色放射线，长度约 32–76 px，越靠近两侧越长，向外流动并循环重置。
//   - 没有停留段，也没有交叉淡化重置；每 4.7 s 重新开始时，树列相位会发生跳变。
// 适用场景: 适合表现速度与沉浸感，或用一次连续运镜飞越整个场所。
// 标签: fpv, drone, fly through, first person, banking, speed, aerial, 镜头, 穿越机, 无人机, 低空, 速度, 沉浸
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit camera -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#035
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 11
// 11 FPV
CELLS[10] = shot((k, w, h, u) => {
  const sp = 14, z = u * sp, T = u * 2 * Math.PI / CYC;
  const c = cam({ z, y: 1.4 + 0.5 * Math.sin(T * 2), roll: 0.22 * Math.sin(T * 2 + 0.6), yaw: 0.05 * Math.sin(T * 2), x: 0.6 * Math.sin(T * 2 - 0.4), pitch: -0.05 });
  const ex = [];
  const L = 8, base = Math.floor(z / L) * L;
  for (let i = 0; i < 12; i++) { const zz = base + i * L; ex.push([-3.6, 0, zz, (kk, s) => tree(kk, s, 4.2)]); ex.push([3.6, 0, zz + L / 2, (kk, s) => tree(kk, s, 4.2, C.blueXL)]); }
  world(k, w, h, c, {
    buildings: false, trees: false, extra: ex, gridX: 20,
    after: (kk) => {   // スピード線
      kk.strokeStyle = 'rgba(150,150,155,.45)'; kk.lineWidth = 2;
      for (let i = 0; i < 10; i++) { const a = i / 10 * Math.PI * 2 + 0.3, ph = mod(u * 2.2 + i * 0.37, 1); const r0 = 120 + ph * 160; kk.beginPath(); kk.moveTo(Math.cos(a) * r0 * 1.9, Math.sin(a) * r0 * 0.8); kk.lineTo(Math.cos(a) * (r0 + 40) * 1.9, Math.sin(a) * (r0 + 40) * 0.8); kk.stroke(); }
    }
  });
  return `${Math.round(sp * 3.6)}km/h`;
});
