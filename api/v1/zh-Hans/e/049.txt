// 模态对话框 / Modal dialog
// 分类: ui (UI 组件) · element 01 · 子类别: 浮层与展开
// 示例表现: 点击“打开”后，整页变暗，一个小窗口在中央略微放大出现；点击 × 将其关闭。
// 动画参数与实现规则:
//   - 在 542×191 的卡片上，周期为 4.5 s。光标在 0.95 s 到达左下方的“打开”按钮，尺寸 120×36，并在 1.0 s 点击。
//   - 打开进度 a：在 1.0–1.32 s 从 0→1，使用 easeOutCubic（三次缓出）。整张卡片覆盖 rgba(18,20,28) 的遮罩，不透明度为 0.42·a。
//   - 窗口：白色，尺寸 280×140、圆角半径 12，中心位于 (271, 97)；阴影模糊半径 24、y 偏移 8、alpha 0.25。缩放比例为 0.9 + 0.1·a，不透明度为 a。
//   - 内部：粗体标题条、两行文字、右下角 88×30 的蓝色按钮，以及右上角的 ×。
//   - 关闭：光标在 3.0 s 前移至 ×，在 3.3 s 点击；a 乘以 1 − easeInOutCubic，作用时段为 3.3–3.58 s。
// 适用场景: 适合需要用户暂停当前操作并处理一件事的场景，例如确认操作或填写简短表单。
// 标签: modal, modal-dialog, dialog, popup, overlay, backdrop, lightbox, confirm, 模态对话框, 弹窗, 遮罩, 确认, 窗口, 交互
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#049
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 01
// 01 モーダル（Modal dialog）
CELLS[0] = ui((w, h, t) => {
  pageLines(24, 18, 494);
  fillRR(24, 92, 120, 36, 8, C.blue); text('打开', 84, 110, 16, '#fff', 600, 'center');
  fillRR(170, 92, 348, 36, 8, C.block); fillRR(24, 142, 494, 38, 8, C.block);
  const a = eo(seg(t, 1.0, 1.32)) * (1 - eio(seg(t, 3.3, 3.58)));
  if (a > 0) {
    X.fillStyle = `rgba(18,20,28,${.42 * a})`; X.fillRect(0, 0, w, h);
    X.save(); X.globalAlpha = a; X.translate(271, 97); const s = .9 + .1 * a; X.scale(s, s);
    shadowed(() => fillRR(-140, -70, 280, 140, 12, '#fff'), 24, 8, .25);
    bar(-116, -48, 120, C.ink, 11); bar(-116, -20, 232); bar(-116, -2, 180);
    fillRR(28, 26, 88, 30, 7, C.blue);
    X.strokeStyle = '#9a9896'; X.lineWidth = 2.2; X.lineCap = 'round'; X.beginPath(); X.moveTo(110, -54); X.lineTo(120, -44); X.moveTo(120, -54); X.lineTo(110, -44); X.stroke();
    X.restore();
  }
  pointer([[.3, w + 30, h + 30], [.95, 86, 114], [1.5, 86, 114], [3.0, 389, 50], [3.5, 389, 50], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
