// 中继器 / Repeater
// 分类: motion2 (动态图形（第 2 弹）) · element 14 · 子类别: 文字与图形
// 示例表现: 一个形状沿圆周复制，各副本旋转并以略有错开的节奏缩放脉动。
// 动画参数与实现规则:
//   - 将一根圆角长条（24 × 60，圆角半径 12）沿画面中心半径为 58 的圆复制，最多 12 份；每份均朝外旋转，角度为其圆周位置角 + 90°。
//   - 副本数量在 0.35–1.7 s 从 1→12，在 3.6–4.45 s 从 12→1，使用二次缓入缓出。新副本从前一个位置滑入自己的位置，向前多转 30°，使用三次缓出。
//   - 整个圆环每 4.7 s 旋转一周。各副本的大小以 1 ± 0.22 的比例脉动，每 4.7 s 两次；每个副本比前一个滞后 0.55 rad，让波纹沿圆环传播。
//   - 颜色按蓝色、珊瑚色、黄色、薄荷绿排列。原始形状带白色描边。右上角显示“×N”计数。
// 适用场景: 适合加载动画、背景图案、花朵或光芒等由单一形状重复组成的画面。
// 标签: repeater, duplicate, radial, array, clone, pattern, loader, 中继器, 重复, 圆环, 加载, 脉动, 图案
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/motion2.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#134
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 14
// 14 リピーター（1つの図形が複製されて円に並び、少しずつずれながら回る）
CELLS[13] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const N = 12, X = w / 2, Y = h / 2, R = 58, spin = 2 * Math.PI * u / CYC;
  const cnt = 1 + (N - 1) * (eIOs(P(u, 0.35, 1.7)) - eIOs(P(u, 3.6, 4.45)));
  for (let i = N - 1; i >= 0; i--) {
    const vis = i === 0 ? 1 : clamp(cnt - i); if (vis <= 0) continue;
    const ang = spin + (i === 0 ? 0 : (i - 1 + eOut(vis))) * 2 * Math.PI / N;
    const pul = (1 + 0.22 * Math.sin(2 * Math.PI * 2 * u / CYC - i * 0.55)) * (0.55 + 0.45 * vis);
    k.save(); k.translate(X + Math.cos(ang) * R, Y + Math.sin(ang) * R); k.rotate(ang + Math.PI / 2); k.scale(pul, pul);
    k.fillStyle = PAL[i % 4]; rr(k, -12, -30, 24, 60, 12); k.fill();
    if (i === 0) { k.strokeStyle = '#fff'; k.lineWidth = 4; rr(k, -12, -30, 24, 60, 12); k.stroke(); }
    k.restore();
  }
  txt(k, '×' + Math.round(cnt), w - 34, 46, `800 40px ${DISP}`, C.yel, 'right', 'middle');
};
