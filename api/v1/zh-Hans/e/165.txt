// 动态形状分隔线 / Animated shape divider
// 分类: lp2 (落地页与网页效果（第 2 弹）) · element 21 · 子类别: 亮点展示
// 示例表现: 蓝色区块与白色区块的交界处，两层波浪形状持续缓慢起伏。
// 动画参数与实现规则:
//   - 视口为 511 × 168：上方为蓝色渐变区块 #3B64E8→#8FA9F7，下方为白色区块；边界处叠加两层波浪。一轮循环为 4.7 s，每个 9.4 s 的效果包周期内执行两次。
//   - 后层波浪为白色，不透明度 0.45，中心线 y = 100；由波长 260 px、振幅 8 px 的正弦波，与波长 130 px、振幅 4 px 的正弦波叠加而成。
//   - 前层波浪为白色，中心线 y = 110；由波长 300 px、振幅 9 px 的波，与波长 110 px、振幅 3.5 px 的波叠加而成，并与下方白色区块相接。
//   - 设相位 a = 2π·t/4.7，后层的两个波分别使用 a + 1 和 −2a，前层分别使用 −a 和 2a + 0.5。每层中的两个正弦波向相反方向传播，使形状不断变化，并每 4.7 s 回到起始状态。
// 适用场景: 适合需要柔和区块边缘的网页，例如海洋、夏日或儿童主题网站。
// 标签: shape-divider, wave, section-divider, svg, ambient, loop, 分隔线, 波浪, 区块, 背景, 形状, 柔和
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit lp2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp2.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#165
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 21
// 21 シェイプディバイダー（セクションの境目の波の形が、ゆっくりゆれ続ける）
CELLS[20] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const a = 2 * Math.PI * u / CYC;
  const wave = (base, A1, L1, p1, A2, L2, p2) => { k.beginPath(); k.moveTo(0, VH + 1); for (let x = 0; x <= VW + 4; x += 4) k.lineTo(x, base + A1 * Math.sin(2 * Math.PI * x / L1 + p1) + A2 * Math.sin(2 * Math.PI * x / L2 + p2)); k.lineTo(VW, VH + 1); k.closePath(); };
  view(k, () => {
    const g = k.createLinearGradient(0, 0, 0, 120); g.addColorStop(0, '#3B64E8'); g.addColorStop(1, '#8FA9F7'); k.fillStyle = g; k.fillRect(0, 0, VW, VH);
    k.fillStyle = 'rgba(255,255,255,.9)'; k.beginPath(); k.arc(430, 44, 22, 0, 7); k.fill();
    txt(k, '夏日优惠活动', 34, 42, `800 22px ${JP}`, '#fff'); btn(k, 34, 56, 96, 24, '了解详情', '#fff', C.blue, 11);
    k.fillStyle = 'rgba(255,255,255,.45)'; wave(100, 8, 260, a + 1, 4, 130, -2 * a); k.fill();
    k.fillStyle = '#fff'; wave(110, 9, 300, -a, 3.5, 110, 2 * a + 0.5); k.fill();
    txt(k, '推荐套餐', 34, 146, `800 15px ${JP}`, C.ink); bars(k, 170, 140, 290, 1, C.faint, 6);
    bars(k, 34, 156, 200, 1, C.faint, 5);
  });
};
