// 滚动监听 / Scrollspy
// 分类: lp2 (落地页与网页效果（第 2 弹）) · element 06 · 子类别: 滚动效果
// 示例表现: 文章滚动时，目录中当前阅读章节的条目会高亮；下一个标题进入指定位置后，高亮随之切换。
// 动画参数与实现规则:
//   - 视口为 511 × 168：左侧目录宽 150 px，含 4 个条目；右侧为文章，标题间距 100 px。滚动距离在 0.3–0.9 s 从 0→100，在 1.35–1.95 s 到 200，在 2.4–3.0 s 到 300，在 3.5–4.4 s 从 300→0，使用三次缓入缓出。一次往返为 4.7 s，每个 9.4 s 的效果包周期内执行两次。
//   - 当前条目对应最近一个顶部位置已越过 scroll + 40 px 的标题，即视口顶部下方 40 px 处的触发线。
//   - 当前条目变化后，在 0.25 s 内使用三次缓出：该条目出现浅橙色背景，尺寸 126 × 26、圆角半径 6；左侧出现宽度为 4 的橙色条，文字从灰色变为橙色粗体。前一个条目在同样的 0.25 s 内恢复原状。
//   - 目录保持固定，只有文章滚动。滚动条滑块随滚动位置变化。
// 适用场景: 适合长文章、文档或常见问题页面，让读者了解当前位于整页的哪个位置。
// 标签: scrollspy, table-of-contents, toc, navigation, active-state, scroll, 滚动监听, 目录, 高亮, 章节, 文档, 导航
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit lp2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp2.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#150
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 06
// 06 スクロールスパイ（左の目次のうち、いま読んでいる見出しの項目が光る）
CELLS[5] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const y = steps([[0.3, 0], [0.9, 100], [1.35, 100], [1.95, 200], [2.4, 200], [3.0, 300], [3.5, 300], [4.4, 0]], u);
  const NAMES = ['开始之前', '使用方法', '价格', '常见问题'], TX = 160;
  const act = t => { const yy = steps([[0.3, 0], [0.9, 100], [1.35, 100], [1.95, 200], [2.4, 200], [3.0, 300], [3.5, 300], [4.4, 0]], t); let a = 0; for (let i = 0; i < 4; i++) if (i * 100 <= yy + 40) a = i; return a; };
  view(k, () => {
    k.save(); k.beginPath(); k.rect(TX - 10, 0, VW, VH); k.clip(); k.translate(0, -y);
    NAMES.forEach((s, i) => {
      const y0 = i * 100 + 14;
      txt(k, `${i + 1}. ${s}`, TX + 14, y0 + 22, `800 16px ${JP}`, C.ink);
      bars(k, TX + 14, y0 + 38, 300, 3, C.faint, 6, 12, 0.6);
    });
    k.restore();
    k.fillStyle = C.soft; k.fillRect(0, 0, TX - 10, VH);
    txt(k, '目录', 20, 30, `800 12px ${JP}`, C.sub);
    const a = act(u);
    NAMES.forEach((s, i) => {
      const q = trans(t => act(t) === i, u, 0.25), iy = 54 + i * 30;
      if (q > 0) { k.fillStyle = `rgba(255,229,219,${q})`; rr(k, 12, iy - 13, TX - 34, 26, 6); k.fill(); k.fillStyle = C.acc; k.globalAlpha = q; rr(k, 12, iy - 13, 4, 26, 2); k.fill(); k.globalAlpha = 1; }
      txt(k, s, 26, iy + 1, `${a === i ? 800 : 600} 13px ${JP}`, mix('#7D879B', C.acc, q), 'left', 'middle');
    });
    sbar(k, y / 300, 0.36);
  });
};
