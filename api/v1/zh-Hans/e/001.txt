// 弹性入场 / Pop-in
// 分类: motion (动态图形) · element 01 · 子类别: 入场与文字
// 示例表现: 应用图标弹跳着出现，稍后通知徽章也随之弹出。
// 动画参数与实现规则:
//   - 每 2.35 s 弹出一次，每个 4.7 s 周期两次：先是珊瑚色底上的星星，再是蓝色底上的爱心。
//   - 图标：116 px 的圆角正方形，圆角半径 30。从 0.28 s 开始，缩放比例按弹簧公式 1 − e^(−6.5t)·cos(15t) 从 0→1，峰值约为 1.26。
//   - 同时，旋转角从 −0.35 rad，约 −20°，按弹簧公式 1 − e^(−6t)·cos(11t) 回到 0。
//   - 徽章：半径 25 的黄色圆形，内部为“1”，位于图标右上方 (+52, −52)。从 0.62 s 开始，按弹簧公式 1 − e^(−7t)·cos(17t) 出现。
//   - 退场：1.86–2.12 s。缩放比例乘以 (1 − backIn, s = 2.2)，使图标先略微膨胀，再缩小至 0。
//   - 地面阴影：62 × 8 的椭圆，随图标缩放而变化，缩放比例限制在 0–1。
// 适用场景: 适合图标、按钮或徽章等小元素的入场，用轻巧的动作为其增加强调。
// 标签: pop, pop-in, spring, bounce, scale, icon, badge, entrance, 弹性, 入场, 弹跳, 图标, 徽章, 缩放
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/motion.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#001
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 01
// 01 ポップイン（アプリのアイコンがポンと弾んで現れ、通知バッジが遅れてポン）
CELLS[0] = (k, w, h, u) => {
  stage(k);
  const T = CYC / 2, n = Math.floor(u / T), lu = u - n * T, X = w / 2, Y = h / 2 + 4, S = 116;
  const col = n % 2 ? C.blue : C.coral;
  const out = backIn(P(lu, 1.86, 2.12), 2.2);
  const sc = spring(lu - 0.28, 15, 6.5) * (1 - out);
  const rot = (1 - spring(lu - 0.28, 11, 6)) * -0.35;
  // 床の影
  k.fillStyle = 'rgba(0,0,0,.28)'; k.beginPath(); k.ellipse(X, Y + 78, 62 * clamp(sc), 8 * clamp(sc), 0, 0, 7); k.fill();
  if (sc > 0.002) {
    k.save(); k.translate(X, Y); k.rotate(rot); k.scale(sc, sc);
    k.fillStyle = col; rr(k, -S / 2, -S / 2, S, S, 30); k.fill();
    k.fillStyle = 'rgba(255,255,255,.16)'; rr(k, -S / 2, -S / 2, S, S / 2, [30, 30, 0, 0]); k.fill();
    k.fillStyle = '#fff';
    if (n % 2) { k.save(); k.translate(0, 2); heartPath(k, 30); k.fill(); k.restore(); }
    else { star(k, 0, 2, 36, 0, 0.48); k.lineJoin = 'round'; k.strokeStyle = '#fff'; k.lineWidth = 6; k.stroke(); k.fill(); }
    k.restore();
    // 通知バッジ
    const bs = spring(lu - 0.62, 17, 7) * (1 - out);
    if (bs > 0.002) { k.save(); k.translate(X + 52 * sc, Y - 52 * sc); k.scale(bs, bs); k.fillStyle = C.yel; k.beginPath(); k.arc(0, 0, 25, 0, 7); k.fill(); k.strokeStyle = C.navy; k.lineWidth = 5; k.stroke(); txt(k, '1', 0, 2, `800 30px ${DISP}`, C.navy, 'center', 'middle'); k.restore(); }
  }
};
