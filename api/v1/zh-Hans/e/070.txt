// 加载转圈 / Spinner
// 分类: ui (UI 组件) · element 22 · 子类别: 导航与状态反馈
// 示例表现: 点击“发送”后，卡片上覆盖浅白色遮罩，中央的蓝色圆弧持续旋转；完成后变成对勾并淡出。
// 动画参数与实现规则:
//   - 卡片尺寸 542×191，周期 4.5 s。光标在 1.0 s 点击底部中央的“发送”按钮，按钮尺寸为 120×40。
//   - 在 1.05–1.2 s 内线性淡入不透明度 0.78 的白色遮罩和加载转圈，遮罩覆盖整个卡片。
//   - 加载转圈位于 (271, 92)：浅色圆环半径 26、粗 6，其上叠加蓝色 243°（1.35π）圆弧，使用圆形线帽；每秒转 1.3 圈，即 468°/s。
//   - 加载转圈在 2.95–3.05 s 淡出；在 2.95–3.2 s 内，半径 28 的蓝色圆形连同白色对勾从 0.7 → 1 缩放，使用 easeOutCubic。
//   - 所有内容在 3.75–4.0 s 内线性淡出。
// 适用场景: 适合发送、保存等持续时间不确定的短暂等待。
// 标签: spinner, loading, loader, activity-indicator, progress-circle, busy, 加载转圈, 旋转, 等待, 发送, 完成反馈, 圆弧
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#070
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 22
// 22 スピナー（Spinner）
CELLS[21] = ui((w, h, t) => {
  const load = seg(t, 1.05, 1.2) * (1 - seg(t, 2.95, 3.05));
  const done = seg(t, 2.95, 3.15) * (1 - seg(t, 3.75, 4.0));
  pageLines(24, 18, 494); strokeRR(24, 90, 494, 34, 7, C.line2, 1.5);
  fillRR(w / 2 - 60, 140, 120, 40, 8, C.blue); text('发送', w / 2, 160, 16, '#fff', 600, 'center');
  const ov = Math.max(load, done);
  if (ov > 0) {
    X.fillStyle = `rgba(255,255,255,${.78 * ov})`; X.fillRect(0, 0, w, h);
    if (load > 0) { X.save(); X.globalAlpha = load; ring(w / 2, 92, 26, '#e3e8f5', 6); X.beginPath(); const a0 = t * Math.PI * 2 * 1.3; X.arc(w / 2, 92, 26, a0, a0 + Math.PI * 1.35); X.strokeStyle = C.blue; X.lineWidth = 6; X.lineCap = 'round'; X.stroke(); X.restore(); }
    if (done > 0) { X.save(); X.globalAlpha = done; const s = .7 + .3 * eo(seg(t, 2.95, 3.2)); X.translate(w / 2, 92); X.scale(s, s); circle(0, 0, 28, C.blue); check(0, 0, 11, '#fff', 4); X.restore(); }
  }
  pointer([[.3, w + 30, h + 30], [.95, w / 2 + 6, 166], [1.5, w / 2 + 6, 166], [2.2, w + 30, h + 30]], [1.0], t);
});
