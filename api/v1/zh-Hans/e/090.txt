// 懒加载 / Lazy loading
// 分类: lp (落地页与网页效果) · element 18 · 子类别: 加载、导航与循环动效
// 示例表现: 灰色占位图滚入视口后，各自出现加载转圈，图片就绪时淡入；第一行从一开始就已加载。
// 动画参数与实现规则:
//   - 占位图组成 3 × 3 网格，每格尺寸 151 × 92；第一行从开始时即显示。滚动位置在 0.35–1.1 s 从 0→110 px，在 1.7–2.5 s 从 110→225 px，在 3.5–4.5 s 从 225→0，使用三次缓入缓出。
//   - 占位图顶边越过视口底部的 168 px 位置时开始加载；每一列比左侧相邻列晚 0.12 s 开始。
//   - 占位图中央显示半径 11 的加载转圈，持续 0.45 s；转圈每 0.8 s 旋转一周。
//   - 随后图片在 0.35 s 内以三次缓出淡入，不透明度从 0→1。尚未加载的占位图显示图片图标。
//   - 从视口底部离开的占位图会重置为隐藏状态。
// 适用场景: 适合图片密集的页面，优先加载首屏可见内容。
// 标签: lazy-loading, lazy-load, images, placeholder, performance, spinner, gallery, 懒加载, 图片加载, 占位图, 加载转圈, 首屏, 按需加载
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#090
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 18
// 18 遅延読み込み（画面に入った画像だけ、その時点で読み込まれて表示される）
CELLS[17] = (k, w, h, u) => {
  frame(k);
  const SY = t => steps([[0.35, 0], [1.1, 110], [1.7, 110], [2.5, 225], [3.5, 225], [4.5, 0]], t), y = SY(u);
  const COLX = [15, 180, 345], IW = 151, IH = 92, ROWS = [12, 168, 276], WAIT = 0.45, FADE = 0.35;
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, '作品画廊', 15, 140, `800 18px ${JP}`, C.ink); bars(k, 120, 128, 200, 1);
    ROWS.forEach((top, r) => COLX.forEach((x, c) => {
      const t0 = r === 0 ? -1e9 : enteredAt(SY, top, VH, u), age = t0 === null ? -1 : u - t0 - c * 0.12;
      k.fillStyle = C.skel; rr(k, x, top, IW, IH, 6); k.fill();
      if (age < WAIT + FADE) {   // まだの画像: 灰色の枠と画像の印
        k.save(); k.translate(x + IW / 2, top + IH / 2); k.strokeStyle = '#B9C1CF'; k.lineWidth = 2; k.lineJoin = 'round';
        if (age >= 0 && age < WAIT) spinner(k, 0, 0, 11, u, C.blue);
        else { rr(k, -15, -11, 30, 22, 3); k.stroke(); k.beginPath(); k.moveTo(-11, 7); k.lineTo(-3, -1); k.lineTo(3, 5); k.lineTo(7, 1); k.lineTo(11, 6); k.stroke(); }
        k.restore();
      }
      if (age >= WAIT) { k.save(); k.globalAlpha = eOut(P(age, WAIT, WAIT + FADE)); pic(k, x, top, IW, IH, (r + c) % 3, 6); k.restore(); }
    }));
    k.restore();
    sbar(k, y / 225, 0.45);
  });
};
