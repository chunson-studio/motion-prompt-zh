// 底部面板 / Bottom sheet
// 分类: ui (UI 组件) · element 03 · 子类别: 浮层与展开
// 示例表现: 手指轻点“查看详情”，面板从底部升起；向下拖动把手即可关闭。
// 动画参数与实现规则:
//   - 在 542×191 的卡片上，周期为 4.5 s。手指以半透明圆形表示，半径 17，带白色边缘；在 1.0 s 轻点底部中央的“查看详情”胶囊按钮，尺寸 170×36。
//   - 打开进度在 1.05–1.45 s 从 0→1，使用 easeOutCubic（三次缓出）。整张卡片覆盖 rgba(18,20,28) 的遮罩，不透明度为 0.36·a。
//   - 面板：白色，高 136，顶部圆角半径 18，顶边位于 y = 191 − 136·a。阴影模糊半径 22、y 偏移 −4；顶部中央有一个 52×6 的拖动把手。
//   - 内部：粗体标题条，以及两行带 26×26 图标的内容。
//   - 关闭：手指在 2.75 s 落到把手 (271, 68)，并在 3.0–3.6 s 向下拖动；a 乘以 (1 − easeInOutCubic，作用于 3.0–3.6 s)，使面板跟随手指移出底部。
// 适用场景: 适合移动端，在保留当前页面背景的同时展示详情或选项。
// 标签: bottom-sheet, sheet, modal-sheet, mobile, slide-up, drag-to-dismiss, panel, 底部面板, 移动端, 手势, 拖动, 详情, 弹出
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#051
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 03
// 03 ボトムシート（Bottom sheet）
CELLS[2] = ui((w, h, t) => {
  pageLines(24, 16, 494); fillRR(24, 76, 239, 54, 8, C.block); fillRR(279, 76, 239, 54, 8, C.block);
  fillRR(186, 144, 170, 36, 18, C.blue); text('查看详情', 271, 162, 15, '#fff', 600, 'center');
  const open = eo(seg(t, 1.05, 1.45));
  const drag = eio(seg(t, 3.0, 3.6));
  const a = open * (1 - drag);
  if (open > 0 && a > 0.001) {
    X.fillStyle = `rgba(18,20,28,${.36 * a})`; X.fillRect(0, 0, w, h);
    const sh = 136, sy = h - sh * a;
    shadowed(() => { rr(0, sy, w, sh + 20, 18); X.fillStyle = '#fff'; X.fill(); }, 22, -4, .2);
    fillRR(w / 2 - 26, sy + 10, 52, 6, 3, C.line2);
    bar(28, sy + 30, 160, C.ink, 11);
    for (let i = 0; i < 2; i++) { const y = sy + 56 + i * 36; fillRR(28, y, 26, 26, 6, i ? C.mb : C.blue); bar(66, y + 4, 220, C.line); bar(66, y + 17, 140, C.line, 7); }
  }
  const sy = h - 136 * a;
  finger([[.3, 271, 162], [1.6, 271, 162], [2.75, w / 2, h - 136 + 13], [3.0, w / 2, h - 136 + 13], [3.6, w / 2, h + 10]], [1.0, 3.0], t, (t > .6 && t < 1.6) || (t > 2.75 && t < 3.6));
});
