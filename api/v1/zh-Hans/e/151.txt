// 回到顶部按钮 / Back-to-top button
// 分类: lp2 (落地页与网页效果（第 2 弹）) · element 07 · 子类别: 滚动效果
// 示例表现: 页面向下滚动后，右下角出现圆形箭头按钮；点击即可平滑返回页顶。
// 动画参数与实现规则:
//   - 视口为 511 × 168。在 0.2–1.7 s 内从 0→300 px 滚动至页面底部，使用三次缓入缓出。一次往返为 4.7 s，每个 9.4 s 的效果包周期内执行两次。
//   - 按钮为右下角的橙色圆形，圆心 x = 472、y = 138，半径 19，带白色向上箭头，阴影模糊半径为 10 px。滚动距离在 80 至 140 px 之间时，按钮上移 10 px 并渐入，使用三次缓出。
//   - 在 2.6 s 点击按钮。鼠标悬停时，按钮在 0.2 s 内从 #FF5A2C 加深为 #D9431C。
//   - 点击后，页面在 2.66–3.26 s 从 300→0 滚回顶部，持续 0.6 s，使用三次缓入缓出；滚动距离降至 80 px 以下时，按钮按相同公式渐出。
// 适用场景: 适合博客长文、长落地页或商品列表，便于读者回到顶部菜单。
// 标签: back-to-top, scroll-to-top, button, navigation, floating-button, smooth-scroll, 回到顶部, 滚动, 按钮, 导航, 长页面, 平滑
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit lp2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp2.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#151
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 07
// 07 トップへ戻るボタン（下までスクロールすると右下に丸いボタンが出て、押すと一気に先頭へ戻る）
CELLS[6] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const CLK = 2.6, MAX = 300;
  const y = u < CLK + 0.02 ? steps([[0.2, 0], [1.7, MAX]], u) : MAX * (1 - eIO(P(u, CLK + 0.06, CLK + 0.66)));
  const cur = t => path([[0, 340, 205], [1.6, 340, 205], [2.35, 472, 138], [2.95, 472, 138], [3.6, 340, 205], [4.7, 340, 205]], t);
  const [cx, cy] = cur(u);
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, 'BLOG', 70, 32, `800 11px ${EN}`, C.acc, 'left', 'alphabetic', '2px');
    txt(k, '周末来一场短途旅行', 70, 58, `800 19px ${JP}`, C.ink);
    pic(k, 70, 72, 371, 100, 2, 8);
    for (let b = 0; b < 3; b++) { const y0 = 192 + b * 82; k.fillStyle = C.ink; rr(k, 70, y0, 150, 9, 4.5); k.fill(); bars(k, 70, y0 + 22, 371, 4, C.faint, 6, 12); }
    k.fillStyle = C.ink; k.fillRect(0, 438, VW, 80); logo(k, 70, 454, '#fff'); bars(k, 300, 451, 140, 1, 'rgba(255,255,255,.3)');
    k.restore();
    const tb = eOut(clamp((y - 80) / 60));
    if (tb > 0) {
      const hov = trans(t => { const [a, b] = cur(t); return Math.hypot(a - 472, b - 138) < 20; }, u, 0.2);
      k.save(); k.globalAlpha = tb; k.translate(0, 10 * (1 - tb));
      k.save(); k.shadowColor = 'rgba(28,35,51,.3)'; k.shadowBlur = 10; k.shadowOffsetY = 3; k.fillStyle = mix(C.acc, '#D9431C', hov); k.beginPath(); k.arc(472, 138, 19, 0, 7); k.fill(); k.restore();
      k.strokeStyle = '#fff'; k.lineWidth = 2.6; k.lineCap = 'round'; k.lineJoin = 'round'; k.beginPath(); k.moveTo(465, 142); k.lineTo(472, 134.5); k.lineTo(479, 142); k.stroke();
      k.restore();
    }
    sbar(k, y / MAX, 0.34);
    click(k, cx, cy, u, [CLK]);
  });
};
