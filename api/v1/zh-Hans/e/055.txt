// 气泡卡片 / Popover
// 分类: ui (UI 组件) · element 07 · 子类别: 浮层与展开
// 示例表现: 点击“详情”后，右侧打开一个带尖角的小白色面板；点击内部的 OK 后关闭。
// 动画参数与实现规则:
//   - 在 542×191 的卡片上，周期为 4.5 s。位置 (24, 112) 放置尺寸 108×36 的蓝色描边“详情”按钮，光标在 1.0 s 点击。
//   - 显示进度 a 在 1.0–1.28 s 从 0→1，使用 easeOutCubic（三次缓出）；不透明度为 a，并从右侧偏移 6 px 的位置滑入。
//   - 面板：白色，尺寸 300×120、圆角半径 10，位于 (156, 62)，边框宽 1 px，阴影模糊半径 22、y 偏移 6。左边缘 y = 122–138 处有一个宽 10 的三角尖角，指向按钮。
//   - 内部：粗体标题条、两行文字、尺寸 70×26 的描边按钮，以及尺寸 72×26 的蓝色 OK 按钮。
//   - 关闭：光标在 3.0 s 前移至 OK 的位置 (402, 157)，在 3.3 s 点击；a 乘以 1 − easeInOutCubic，作用时段为 3.3–3.5 s。
// 适用场景: 适合紧邻触发按钮展示简短设置或确认内容。
// 标签: popover, popup, callout, bubble, flyout, overlay, 气泡卡片, 弹出, 面板, 详情, 确认, 按钮
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#055
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 07
// 07 ポップオーバー（Popover）
CELLS[6] = ui((w, h, t) => {
  pageLines(24, 16, 494);
  strokeRR(24, 112, 108, 36, 8, C.blue, 2); text('详情', 78, 130, 15, C.blue, 600, 'center');
  fillRR(24, 160, 494, 22, 6, C.block);
  const a = eo(seg(t, 1.0, 1.28)) * (1 - eio(seg(t, 3.3, 3.5)));
  if (a > 0) {
    X.save(); X.globalAlpha = a; X.translate(6 * (1 - a), 0);
    const px = 156, py = 62, pw = 300, ph = 120;
    shadowed(() => { fillRR(px, py, pw, ph, 10, '#fff'); }, 22, 6, .2); strokeRR(px, py, pw, ph, 10, C.border, 1);
    X.beginPath(); X.moveTo(px + 1, 122); X.lineTo(px - 10, 130); X.lineTo(px + 1, 138); X.fillStyle = '#fff'; X.fill();
    X.beginPath(); X.moveTo(px, 122); X.lineTo(px - 10, 130); X.lineTo(px, 138); X.strokeStyle = C.border; X.lineWidth = 1; X.stroke();
    bar(px + 18, py + 18, 130, C.ink, 11); bar(px + 18, py + 42, 250); bar(px + 18, py + 60, 200);
    strokeRR(px + 130, py + 82, 70, 26, 6, C.line2, 1.5); fillRR(px + 210, py + 82, 72, 26, 6, C.blue); text('OK', px + 246, py + 95, 14, '#fff', 700, 'center', NUMF);
    X.restore();
  }
  pointer([[.3, w + 30, h + 30], [.95, 84, 134], [1.6, 84, 134], [3.0, 402, 157], [3.5, 402, 157], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
