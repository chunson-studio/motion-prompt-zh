// 分类: lp (落地页与网页效果) · element 22 · 子类别: 加载、导航与循环动效
// 示例表现: 深蓝色横条中，大号服务名称由橙色圆点分隔，持续无缝地向左滚动。
// 动画参数与实现规则:
//   - 横条高 58，深蓝色，占满视口宽度。文字为白色，字号 30 px，使用重字重，字距 1 px。
//   - 一组包含四个词组：WEB DESIGN / BRANDING / MOTION / UI/UX；相隔 64 px，每个间隙内有一个半径 6 的橙色圆点。
//   - 每个周期为 4.8 s，以匀速向左移动一整组的宽度，此处约为 881 px。先测量整组宽度，再以此宽度对位移取模，因此没有可见接缝。
//   - 重复整组内容，直到填满视口宽度。
// 适用场景: 适合富有节奏地展示服务名称、客户标志或宣传语。
// 标签: marquee, ticker, infinite-loop, scrolling-text, logo-strip, band, loop, 跑马灯, 循环, 横向滚动, 服务名称, 无缝, 文字动画
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#094
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 22
// 22 マーキー（文字の帯が、途切れずに横へ流れ続ける）
CELLS[21] = (k, w, h, u) => {
  frame(k);
  const WORDS = ['WEB DESIGN', 'BRANDING', 'MOTION', 'UI/UX'], F = `800 30px ${EN}`, GAPW = 64;
  view(k, () => {
    bars(k, 34, 20, 180, 1, C.ink, 10); bars(k, 34, 38, 300, 1);
    k.fillStyle = C.ink; k.fillRect(0, 58, VW, 58);
    k.font = F; k.letterSpacing = '1px'; const ws = WORDS.map(s => k.measureText(s).width); k.letterSpacing = '0px';
    const unit = ws.reduce((a, b) => a + b + GAPW, 0);
    let x = -mod(u / CYC, 1) * unit;                 // 1周期で1組ぶん左へ
    for (let rep = 0; x < VW; rep++) WORDS.forEach((s, i) => {
      txt(k, s, x, 88, F, '#fff', 'left', 'middle', '1px');
      k.fillStyle = C.acc; k.beginPath(); k.arc(x + ws[i] + GAPW / 2, 88, 6, 0, 7); k.fill();
      x += ws[i] + GAPW;
    });
    for (let i = 0; i < 3; i++) { k.fillStyle = C.soft; rr(k, 34 + i * 152, 130, 136, 30, 6); k.fill(); bars(k, 46 + i * 152, 142, 90, 1, C.faint); }
  });
};
