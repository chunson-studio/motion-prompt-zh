// 筛选标签 / Filter chips
// 分类: ui (UI 组件) · element 13 · 子类别: 切换与输入
// 示例表现: 点击“最新”和“免费”筛选标签后，标签出现对勾并变蓝，不符合条件的卡片随之变淡。
// 动画参数与实现规则:
//   - 在 542×191 的卡片上，周期为 4.5 s。四个标签分别为全部、最新、热门、免费；高 34、圆角半径 17，宽度为文字宽度 + 32，从左侧起排列，间隔 10 px。
//   - 点击：在 0.95 s 开启最新，在 1.75 s 开启免费，在 3.0 s 关闭最新，在 3.6 s 关闭免费。
//   - 每次变化持续 0.15 s，使用 easeOutCubic（三次缓出）；标签增宽 18 px，为对勾腾出位置，填充变为浅蓝色，边框和文字变为蓝色。
//   - 下方列表包含四张卡片，尺寸 242×54，分两列两行；不符合条件的卡片不透明度从 1 降至 0.12。
// 适用场景: 适合组合几个快捷筛选条件，缩小列表或搜索结果范围。
// 标签: chips, filter-chips, tags, filter, facets, pills, 筛选标签, 筛选, 组合条件, 列表, 对勾, 搜索
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#061
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 13
// 13 チップ（Filter chips）
CELLS[12] = ui((w, h, t) => {
  const chips = ['全部', '最新', '热门', '免费']; let x = 24; const pos = [];
  const on = [0, eo(seg(t, 0.95, 1.1)) * (1 - eo(seg(t, 3.0, 3.15))), 0, eo(seg(t, 1.75, 1.9)) * (1 - eo(seg(t, 3.6, 3.75)))];
  chips.forEach((s, i) => {
    const k = on[i], cw0 = tw(s, 15, 600) + 32 + (k > 0 ? 18 * k : 0);
    fillRR(x, 12, cw0, 34, 17, k > .5 ? C.lb : '#fff'); strokeRR(x, 12, cw0, 34, 17, k > .5 ? C.blue : C.line2, 1.5);
    if (k > 0) { X.globalAlpha = k; check(x + 18, 29, 5, C.blue, 2.4); X.globalAlpha = 1; }
    text(s, x + 16 + 18 * k, 30, 15, k > .5 ? C.blue : '#555', 600);
    pos.push(x + cw0 / 2); x += cw0 + 10;
  });
  // 一覧：条件に合わないカードが消える
  const vis = [1, 1 - Math.max(on[1], on[3]), 1 - on[3], 1];
  for (let i = 0; i < 4; i++) {
    const cx = 24 + (i % 2) * 252, cy = 60 + Math.floor(i / 2) * 64, a = clamp(.12 + .88 * vis[i]);
    X.save(); X.globalAlpha = a; strokeRR(cx, cy, 242, 54, 8, C.line2, 1.5); fillRR(cx + 10, cy + 10, 34, 34, 6, i === 0 ? C.mb : C.line); bar(cx + 54, cy + 14, 120, C.line2, 10); bar(cx + 54, cy + 32, 150, C.line, 7); X.restore();
  }
  const p1 = pos[1] || 120, p3 = pos[3] || 300;
  pointer([[.3, w + 30, h + 30], [.9, 130, 32], [1.3, 130, 32], [1.7, 300, 32], [2.5, 300, 32], [2.95, 130, 32], [3.2, 130, 32], [3.55, 296, 32], [3.85, 296, 32], [4.3, w + 30, h + 30]], [0.95, 1.75, 3.0, 3.6], t);
});
