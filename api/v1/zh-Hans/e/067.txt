// 巨型菜单 / Mega menu
// 分类: ui (UI 组件) · element 19 · 子类别: 导航与状态反馈
// 示例表现: 将光标悬停在页头的“产品”上，会展开通栏面板，内含三列链接和一张图片；光标移开后收起。
// 动画参数与实现规则:
//   - 卡片尺寸 542×191，周期 4.5 s。页头高 42，显示“产品、价格、支持”，字号 16 px，每项都有向下箭头。
//   - 光标仅悬停，不点击：1.0 s 到达“产品”。展开量 a 在 1.05–1.3 s 从 0→1，使用 easeOutCubic；“产品”变蓝并显示下划线。
//   - 面板为白色、通栏宽、高 146；随 a 淡入，并向下移动 12 px，在页头下方就位。阴影模糊为 18，y 偏移为 8。
//   - 内部三列相隔 130 px，每列含一个蓝色标题和三条链接；右侧图片尺寸为 104×112。光标向下移动，第二列的第二条链接在 1.75 至 2.9 s 以浅蓝色高亮。
//   - 关闭：光标从 2.9 s 开始离开面板，面板在 3.15–3.4 s 淡出，使用 1 − easeInOutCubic。
// 适用场景: 适合产品或栏目较多的网站，让访客一次看到完整的导航范围。
// 标签: mega-menu, megamenu, dropdown-navigation, navigation, header-menu, multi-column, 巨型菜单, 导航, 悬停, 通栏, 产品分类, 展开面板
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#067
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 19
// 19 メガメニュー（Mega menu）
CELLS[18] = ui((w, h, t) => {
  const a = eo(seg(t, 1.05, 1.3)) * (1 - eio(seg(t, 3.15, 3.4)));
  pageLines(24, 62, 494); fillRR(24, 130, 494, 50, 8, C.block);
  X.fillStyle = '#fff'; X.fillRect(0, 0, w, 42); X.fillStyle = C.line; X.fillRect(0, 42, w, 1.5);
  fillRR(24, 14, 70, 16, 4, C.mb);
  const nav = [['产品', 190], ['价格', 262], ['支持', 334]];
  nav.forEach(([s, x], i) => { const on = i === 0 && a > .3; text(s, x, 22, 16, on ? C.blue : '#555', on ? 700 : 600); if (on) fillRR(x, 36, tw(s, 16, 700), 3, 1.5, C.blue); chevron(x + tw(s, 16, 700) + 10, 22, 3.5, 0, on ? C.blue : '#999', 1.8); });
  if (a > 0) {
    X.save(); X.globalAlpha = a; X.beginPath(); X.rect(0, 43, w, h); X.clip(); const py = 43 - 12 * (1 - a);
    shadowed(() => { X.fillStyle = '#fff'; X.fillRect(0, py, w, 146); }, 18, 8, .16);
    for (let c = 0; c < 3; c++) { const cx = 24 + c * 130; bar(cx, py + 16, 80, C.blue, 10); for (let k = 0; k < 3; k++) { const hov = c === 1 && k === 1 && t > 1.75 && t < 2.9; if (hov) fillRR(cx - 6, py + 34 + k * 26, 112, 22, 5, C.lb); bar(cx, py + 40 + k * 26, [90, 70, 100][(k + c) % 3], hov ? C.sb : C.line2, 9); } }
    mountains(414, py + 14, 104, 112);
    X.restore();
  }
  // 乗せるだけで開く（クリックしない）
  pointer([[.3, w + 30, h + 30], [1.0, 205, 26], [1.4, 205, 26], [1.8, 196, 98], [2.9, 196, 98], [3.2, 300, 200], [3.9, w + 30, h + 30]], [], t);
});
