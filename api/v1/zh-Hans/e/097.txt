// 逆光 / Backlighting
// 分类: photo (摄影效果、光线与风格) · element 01 · 子类别: 布光与时段
// 示例表现: 太阳从画外移动到人物头部正后方；人物边缘出现细亮线，画面覆盖浅暖色薄雾，朝向镜头的一侧变暗。
// 动画参数与实现规则:
//   - 场景 1：人物半身像位于 y 150 的地平线前方，头部圆心为 x 270、y 78，半径 29 px，不绘制面部或头发。光照采用平涂琥珀色 #FFC75A，阴影采用平涂色 #33405A。
//   - 太阳半径 20 px，从画外左上方 (x 40, y −40) 移至头部正后方 (x 270, y 56)；附带半径为太阳 2.6× 和 1.7× 的平涂光环，颜色 #FFE3A3，不透明度分别为 35% 和 60%。
//   - 轮廓：先用粗 7 px 的线描边，再填充人物内部，在外侧留下 3.5 px 的亮边。
//   - 人物颜色从 #4D5A75 → #33405A；天空从 #FAF9F7 → #FFF4DC；在整个画面叠加不透明度 22% 的 #FFE3A3。
//   - 时序：效果在 0.4 s 至 1.4 s 进入，保持到 4.0 s，并在 4.8 s 前完全消失；单次周期 = 5 s，使用三次缓入缓出，约等于 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合轻盈、带发光感的人像，或希望表现被光线包围的画面。
// 标签: backlighting, backlit, contre-jour, sun behind, haze, glow, halo, 逆光, 轮廓, 太阳, 人像, 暖光, 薄雾
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#097
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 01
// 逆光: 太陽が画面の外から頭の真後ろへ回り込み、輪郭が光って人物は暗くなる
CELLS[0] = cell((k, u) => {
  const a = env(u);
  const BACK = Object.assign({}, BASE, { sky: '#FFF4DC', sky2: '#FFEBC4', far: '#F6E2BE', ground: '#F7EAD3', bld: '#F7E9D2', winOff: '#EEDCBF', win: '#EEDCBF', tree: '#EEDCBF', edge: '#E3CFAE', post: '#CDB894', trunk: '#E3CFAE', bulb: '#E3CFAE' });
  stagePortrait(k, pal(BASE, BACK, a), { sun: [lerp(40, PX, a), lerp(-40, PY - 22, a), 20, 1] });
  // 輪郭の光: 胸像の形を太い線でなぞってから中を塗り、縁だけを残す
  k.save(); k.globalAlpha = a; bustPath(k, PX, PY, PR); k.strokeStyle = BASE.lit; k.lineWidth = 7; k.lineJoin = 'round'; k.stroke(); k.restore();
  bust(k, PX, PY, PR, mixc(BASE.person, BASE.shade, a));
  // かすみ: 全体に光の色を薄く重ねる
  k.save(); k.globalAlpha = 0.22 * a; k.fillStyle = BASE.glow; k.fillRect(0, 0, CW, CH); k.restore();
});
