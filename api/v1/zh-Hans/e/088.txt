// 离站挽留弹窗 / Exit-intent popup
// 分类: lp (落地页与网页效果) · element 16 · 子类别: 悬停与点击交互
// 示例表现: 光标从页面顶部向标签页或关闭按钮移出时，页面变暗并显示“稍等一下！”弹窗；点击 × 可关闭。
// 动画参数与实现规则:
//   - 光标在 0.7–1.25 s 从页面内向浏览器地址栏移动，使用三次缓入缓出。
//   - 光标越过视口顶边时触发，即 y = 0，约在 1.07 s；判断方式类似 document 的 mouseleave 事件中检查 clientY ≤ 0。
//   - 触发后 0.05–0.45 s 展开，使用三次缓出：深蓝色背景遮罩的不透明度从 0→0.5；弹窗尺寸 260 × 132、圆角半径 12，不透明度从 0→1，缩放比例从 0.9→1。
//   - 弹窗内包含标题、橙色优惠说明、优惠券按钮和右上角的 ×。
//   - 在 2.9 s 点击 ×；随后在 0.03–0.25 s 内线性消失。
// 适用场景: 适合向即将离开、尚未转化的访客展示最后一次优惠信息。
// 标签: exit-intent, popup, mouseleave, coupon, conversion, retention, lp, 离站弹窗, 挽留, 优惠券, 弹窗, 光标离开, 营销
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#088
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 16
// 16 離脱防止ポップアップ（カーソルがタブや閉じるボタンの方へ出ていくと、引き止める小窓が出る）
CELLS[15] = (k, w, h, u) => {
  frame(k);
  const OX = WIN.x, OY = WIN.y + WIN.bar;               // カーソルは窓の外（タブ・アドレスバー）まで出るので、カードの座標で描く
  const [cx, cy] = path([[0, 440, 150], [0.7, 300, 120], [1.25, 360, -18], [1.9, 372, -20], [2.6, 366, 38], [3.3, 366, 38], [4.4, 440, 150], [4.8, 440, 150]], u);
  const CLOSE = 2.9;
  let TRIG = 0.7; while (TRIG < 1.3 && path([[0.7, 300, 120], [1.25, 360, -18]], TRIG)[1] >= 0) TRIG += 1 / 240;   // 表示域の上端（y = 0）を越えた時刻（約1.07秒）
  const m = eOut(P(u, TRIG + 0.05, TRIG + 0.45)) * (1 - P(u, CLOSE + 0.03, CLOSE + 0.25));
  const D = { x: 125.5, y: 18, w: 260, h: 132 };
  view(k, () => {
    header(k, 0);
    pic(k, 24, 46, 222, 110, 0, 8); txt(k, '给初次到访的你', 264, 82, `800 19px ${JP}`, C.ink); bars(k, 264, 96, 210, 3); btn(k, 264, 132, 100, 24, '申请');
    if (m > 0) {
      k.fillStyle = `rgba(28,35,51,${0.5 * m})`; k.fillRect(0, 0, VW, VH);
      k.save(); k.globalAlpha = m; k.translate(D.x + D.w / 2, D.y + D.h / 2); const s = 0.9 + 0.1 * m; k.scale(s, s); k.translate(-D.w / 2, -D.h / 2);
      k.fillStyle = '#fff'; rr(k, 0, 0, D.w, D.h, 12); k.fill();
      txt(k, '稍等一下！', D.w / 2, 34, `800 17px ${JP}`, C.ink, 'center');
      txt(k, '首单立减 10%', D.w / 2, 60, `800 15px ${JP}`, C.acc, 'center');
      btn(k, D.w / 2 - 75, 80, 150, 30, '领取优惠券', C.acc, '#fff', 12);
      k.strokeStyle = C.sub; k.lineWidth = 2; k.lineCap = 'round'; k.beginPath(); k.moveTo(D.w - 26, 14); k.lineTo(D.w - 14, 26); k.moveTo(D.w - 14, 14); k.lineTo(D.w - 26, 26); k.stroke();
      k.restore();
    }
  });
  // カーソルの × のクリック（窓の中）と、窓の外への移動
  k.save(); k.beginPath(); k.rect(0, 0, CW, CH); k.clip(); k.translate(OX, OY); click(k, cx, cy, u, [CLOSE]); k.restore();
};
