// 滚动放大 / Zoom on scroll
// 分类: lp2 (落地页与网页效果（第 2 弹）) · element 08 · 子类别: 滚动效果
// 示例表现: 页面滚动时，中央小照片逐渐放大至填满窗口，周围文字消退，照片上浮现标题。
// 动画参数与实现规则:
//   - 视口为 511 × 168。滚动进度 p 在 0.3–2.3 s 从 0→1，在 3.0–4.4 s 从 1→0，使用三次缓入缓出。一次往返为 4.7 s，每个 9.4 s 的效果包周期内执行两次。
//   - 照片框从中央的 155 × 90，位置 x = 178、y = 58、圆角半径 8，放大到完整的 511 × 168 窗口，圆角半径变为 0；位置、宽度、高度和圆角半径均与 p 成比例变化。
//   - 当 p 从 0→0.45 时，周围的标题和文字渐出，不透明度为 1 − 2.2p。
//   - 当 p 从 0.7→1 时，照片中央渐入白色标题，字号 26 px、字距 2 px。滚动条滑块随滚动位置变化。
// 适用场景: 适合旅游、房地产或品牌网站，将访客的注意力逐渐引入照片。
// 标签: zoom, scroll, image, scale, scroll-driven, hero, fullscreen, 滚动, 放大, 照片, 全屏, 沉浸, 品牌
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit lp2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp2.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#152
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 08
// 08 スクロールで拡大（スクロールに合わせて、小さな写真が窓いっぱいまで広がる）
CELLS[7] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const p = steps([[0.3, 0], [2.3, 1], [3.0, 1], [4.4, 0]], u);
  const x = lerp(178, 0, p), y = lerp(58, 0, p), ww = lerp(155, VW, p), hh = lerp(90, VH, p);
  view(k, () => {
    const a = 1 - clamp(p * 2.2);
    k.save(); k.globalAlpha = a;
    txt(k, '走进风景里', VW / 2, 38, `800 19px ${JP}`, C.ink, 'center');
    bars(k, 34, 66, 120, 4, C.faint, 6, 14); bars(k, 357, 66, 120, 4, C.faint, 6, 14);
    k.restore();
    pic(k, x, y, ww, hh, 0, lerp(8, 0, p));
    const b = clamp((p - 0.7) / 0.3);
    if (b > 0) { k.save(); k.globalAlpha = b; txt(k, '欢迎来到绝美之境', VW / 2, VH / 2 + 2, `800 26px ${JP}`, '#fff', 'center', 'middle', '2px'); k.restore(); }
    sbar(k, p * 0.6, 0.4);
  });
};
