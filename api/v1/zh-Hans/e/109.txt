// 镜头眩光 / Lens flare
// 分类: photo (摄影效果、光线与风格) · element 13 · 子类别: 镜头与相机效果
// 示例表现: 太阳从建筑后方露出时，周围出现光芒；光点、六边形和光环沿太阳穿过画面中心的连线，延伸到对侧。
// 动画参数与实现规则:
//   - 场景 2：街角中，建筑、树木和路灯（x 114、292、494）位于 y 132–162 的道路后方；人物站在左前方，脚底位于 x 176、y 200，身高 124 px。
//   - 太阳半径 13 px，从建筑后方 (x 428, y 74) 移至开阔天空 (x 474, y 30)。
//   - 光芒共 8 条，从距太阳中心 17 px 处向外延伸，终点交替为 26 px 或 44 px，即长度分别为 9 px 和 27 px；琥珀色，粗 1.6 px，不透明度 75%。
//   - 眩光位于太阳穿过画面中心的直线上，位置参数 t 依次为 0.45、0.8、1.15、1.45、1.75、2.0，其中 t = 1 为画面中心；依次绘制 7 px 圆形、13 px 六边形、5 px 圆形、22 px 六边形、9 px 圆形和 30 px 圆环，圆环线宽 3 px。
//   - 颜色使用琥珀色 #FFC75A 和 #FFE3A3，不透明度为 40–70% × 效果强度。
//   - 时序：效果在 0.4 s 至 1.4 s 进入，保持到 4.0 s，并在 4.8 s 前完全消失；单次周期 = 5 s，使用三次缓入缓出，约等于 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合增加耀眼光感、夏日阳光或电影式光线氛围。
// 标签: lens flare, flare, ghosting, sun flare, starburst, cinematic, 镜头眩光, 光斑, 太阳, 光芒, 电影感, 夏日
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#109
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 13
// レンズフレア: 建物の陰から太陽が出ると、画面の中心をはさんで反対側へ光の玉が並ぶ
CELLS[12] = cell((k, u) => {
  const a = env(u), sx = lerp(428, 474, a), sy = lerp(74, 30, a);
  stageStreet(k, BASE, { sun: [sx, sy, 13, 1] });
  const cx = CW / 2, cy = CH / 2, dx = cx - sx, dy = cy - sy;
  k.save();
  // 太陽の光の筋
  k.globalAlpha = 0.75 * a; k.strokeStyle = BASE.lit; k.lineWidth = 1.6; k.lineCap = 'round';
  for (let i = 0; i < 8; i++) { const t = i / 8 * TAU + 0.2, L = i % 2 ? 26 : 44; k.beginPath(); k.moveTo(sx + Math.cos(t) * 17, sy + Math.sin(t) * 17); k.lineTo(sx + Math.cos(t) * L, sy + Math.sin(t) * L); k.stroke(); }
  // 光の玉: 太陽から中心を通る線の上（t = 0.45〜2.0）
  const G = [[0.45, 7, 'c', 0.6], [0.8, 13, 'h', 0.55], [1.15, 5, 'c', 0.7], [1.45, 22, 'h', 0.4], [1.75, 9, 'c', 0.55], [2.0, 30, 'r', 0.55]];
  for (const [t, r, kind, al] of G) {
    const x = sx + dx * t, y = sy + dy * t;
    k.globalAlpha = al * a;
    if (kind === 'r') { k.strokeStyle = BASE.lit; k.lineWidth = 3; k.beginPath(); k.arc(x, y, r, 0, TAU); k.stroke(); continue; }
    k.fillStyle = kind === 'h' ? BASE.glow : BASE.lit; k.beginPath();
    if (kind === 'h') for (let i = 0; i < 6; i++) { const t2 = i / 6 * TAU; k.lineTo(x + Math.cos(t2) * r, y + Math.sin(t2) * r); } else k.arc(x, y, r, 0, TAU);
    k.closePath(); k.fill();
  }
  k.restore();
});
