// 抽屉面板 / Drawer (Off-canvas)
// 分类: ui (UI 组件) · element 02 · 子类别: 浮层与展开
// 示例表现: 点击“菜单”后，高面板从左侧滑入；点击外部区域后滑回。
// 动画参数与实现规则:
//   - 在 542×191 的卡片上，周期为 4.5 s。光标在 1.0 s 点击左上角的“菜单”按钮，尺寸 110×34。
//   - 打开进度 a：在 1.0–1.38 s 从 0→1，使用 easeOutCubic（三次缓出）。整张卡片覆盖 rgba(18,20,28) 的遮罩，不透明度为 0.36·a。
//   - 面板：白色，宽 210，高度与卡片一致；x = −210 + 210·a，因此从左边缘滑入。阴影模糊半径 22，alpha 0.25。
//   - 内部：顶部为半径 14 的头像和姓名，下方为四行菜单，间隔 32 px；第一行使用浅蓝色背景带表示选中。
//   - 关闭：在 3.3 s，光标点击外部位置 (420, 120)；a 乘以 1 − easeInOutCubic，作用时段为 3.3–3.65 s，面板向左滑回。
// 适用场景: 适合将导航或筛选项收在屏幕外，按需展开，尤其适用于移动端。
// 标签: drawer, off-canvas, offcanvas, side-menu, navigation-drawer, sheet, sidebar, slide-in, 抽屉, 侧栏, 菜单, 导航, 滑入, 移动端
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#050
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 02
// 02 ドロワー（Drawer (Off-canvas)）
CELLS[1] = ui((w, h, t) => {
  strokeRR(24, 14, 110, 34, 8, C.line2, 1.5); text('菜单', 79, 31, 15, '#555', 600, 'center');
  bar(160, 26, 140, C.line2, 11); pageLines(24, 70, 494); fillRR(24, 138, 494, 42, 8, C.block);
  const a = eo(seg(t, 1.0, 1.38)) * (1 - eio(seg(t, 3.3, 3.65)));
  if (a > 0) {
    X.fillStyle = `rgba(18,20,28,${.36 * a})`; X.fillRect(0, 0, w, h);
    const pw = 210, px = -pw + pw * a;
    shadowed(() => { X.fillStyle = '#fff'; X.fillRect(px, 0, pw, h); }, 22, 0, .25);
    circle(px + 34, 32, 14, C.mb); bar(px + 58, 27, 100, C.line2, 10);
    for (let i = 0; i < 4; i++) { const y = 62 + i * 32; if (i === 0) fillRR(px + 12, y - 2, pw - 24, 28, 6, C.lb); fillRR(px + 22, y + 5, 14, 14, 3, i === 0 ? C.blue : C.line2); bar(px + 46, y + 7, 90 + (i % 2) * 30, i === 0 ? C.sb : C.line); }
  }
  pointer([[.3, w + 30, h + 30], [.95, 70, 34], [1.6, 70, 34], [3.0, 420, 120], [3.6, 420, 120], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
