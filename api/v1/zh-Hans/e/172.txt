// 长镜头 / Long take
// 分类: camera2 (镜头运动与转场（第 2 弹）) · element 04 · 子类别: 镜头运动
// 示例表现: 一个连续镜头跟随行走的人物，全程没有剪切；底部时间条保持为完整的一段。
// 动画参数与实现规则:
//   - 衔接：没有剪切，一个镜头连续跟随人物 4.2 s；#041 跳切则将过程分为五段。
//   - 主体：人物以恒定速度从左前方走向右后方，起点 x = -3.4 m、纵深 5.2 m，终点 x = 2.8 m、纵深 9.6 m；四肢摆动角度为 sin(t×7)×0.42 rad。
//   - 相机：跟随人物向右横移 0 m→1.4 m，同时转向，偏航角 = atan2(person x - camera x, depth)×0.8；相机高度 1.4 m。
//   - 底部时间条：一整条蓝色横条，标有“1镜到底·无剪切”，并显示播放头，用于说明镜头没有中断。
//   - 返回：在 4.2 s 停止，再在最后 0.5 s 交叉淡化回起始状态，一次演示为 4.7 s。
//   - 右上角读数显示已过时间：NO CUT 0.0s→4.2s。
// 适用场景: 适合保持紧张感与现场感的连续性，或将一连串动作呈现为完整流畅的过程。
// 标签: long take, oner, one shot, continuous, no cut, sequence shot, camera move, 镜头, 长镜头, 连续, 跟随, 一镜到底, 无剪切
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit camera2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera2.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#172
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 04
// 04 長回し: 歩く人物をカットなしで追い続ける1ショット。画面下の帯は1本のまま（#041 ジャンプカットのように割れない）
CELLS[3] = shot((k, w, h, u) => {
  const uu = Math.min(u, CYC - RESET);
  const draw = (kk, t) => {
    const p = t / (CYC - RESET);
    const x = lerp(-3.4, 2.8, p), z = lerp(5.2, 9.6, p), cx = lerp(0, 1.4, p);
    world(kk, w, h, cam({ x: cx, y: 1.4, yaw: Math.atan2(x - cx, z) * 0.8 }), { person: { x, z, walk: t * 7 }, fg: [[-2.6, 3.4], [3.6, 4.6, 'rock']] });
    strip(kk, w, h, [[0, CYC, '1镜到底·无剪切', C.blue]], t);
  };
  const r = P(u, CYC - RESET, CYC);
  blend(k, kk => draw(kk, uu), kk => draw(kk, 0), eIOs(r));
  return `NO CUT ${(r > .5 ? 0 : uu).toFixed(1)}s`;
});
