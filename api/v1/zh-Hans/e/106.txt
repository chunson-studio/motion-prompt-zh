// 移轴效果 / Tilt-shift
// 分类: photo (摄影效果、光线与风格) · element 10 · 子类别: 镜头与相机效果
// 示例表现: 街道顶部和底部变模糊，仅中间包含道路、汽车和小人物的横向区域保持清晰，使街景看起来像模型。
// 动画参数与实现规则:
//   - 街角包含位于 y 132–162 道路后方的建筑、树木和路灯，不绘制大型前景人物；四个身高 23–26 px 的小人物站在人行道上，脚底位于 y 172；汽车每 5 s 从 x −60 行驶到 600。
//   - 合焦带：y 66–176 保持清晰。
//   - 模糊：y 0–42 和 y 190–208 的模糊半径从 0 → 7 px；y 42–66 和 y 176–190 从 0 → 3 px。
//   - 合焦带边缘绘制虚线，线段长 6 px、间隔 5 px，颜色 #8E8E93，不透明度 75%。
//   - 时序：效果在 0.4 s 至 1.4 s 进入，保持到 4.0 s，并在 4.8 s 前完全消失；单次周期 = 5 s，使用三次缓入缓出，约等于 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合将城市或活动现场表现成玩具模型世界。
// 标签: tilt-shift, miniature, miniature effect, fake miniature, selective focus, toy town, focus band, 移轴, 微缩, 模型感, 景深, 模糊带, 城市
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#106
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 10
// ティルトシフト: 上下の帯がぼけて中央の帯だけピントが合い、街と小さな人と車が模型のように見える
CELLS[9] = cell((k, u) => {
  const a = env(u), carX = lerp(-60, 600, u / CYC);
  const minis = D => { [[196, 26], [322, 26], [340, 23], [452, 25]].forEach(([x, h]) => figure(D, x, 172, h, BASE.person)); };
  const [S, ks] = layer('tilt'); stageStreet(ks, BASE, { car: carX, person: false, after: minis });
  const [B1, k1] = layer('tilt-1'); k1.filter = `blur(${(3 * a).toFixed(2)}px)`; k1.drawImage(S, 0, 0); k1.filter = 'none';
  const [B2, k2] = layer('tilt-2'); k2.filter = `blur(${(7 * a).toFixed(2)}px)`; k2.drawImage(S, 0, 0); k2.filter = 'none';
  k.drawImage(S, 0, 0);
  const band = (img, y0, y1) => { k.save(); k.beginPath(); k.rect(0, y0, CW, y1 - y0); k.clip(); k.drawImage(img, 0, 0); k.restore(); };
  band(B2, 0, 42); band(B1, 42, 66); band(B1, 176, 190); band(B2, 190, CH);
  // ピントの帯の目印（写真アプリの線）
  k.save(); k.globalAlpha = 0.75 * a; k.strokeStyle = '#8E8E93'; k.lineWidth = 1.4; k.setLineDash([6, 5]);
  k.beginPath(); k.moveTo(0, 66); k.lineTo(CW, 66); k.moveTo(0, 176); k.lineTo(CW, 176); k.stroke(); k.setLineDash([]);
  k.restore();
});
