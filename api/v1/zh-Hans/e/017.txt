// 遮罩转场 / Mask transition
// 分类: motion (动态图形) · element 17 · 子类别: 转场与强调
// 示例表现: 圆形或星形视窗逐渐扩大，在内部显露下一个场景。
// 动画参数与实现规则:
//   - 白昼场景→夜景→白昼场景，每段 2.35 s；每半周期的 0.45–1.6 s 执行转场，持续 1.15 s，使用三次缓入缓出。
//   - 将下一个场景绘制在当前场景上方，并裁切到指定形状内；该形状不断扩大，直至覆盖整个画面。
//   - 白昼→夜景：从太阳中心 (112, 104) 展开圆形，半径从 0→470。
//   - 夜景→白昼：从月亮中心 (427, 104) 展开五角星，外半径从 0→1150，扩大的同时旋转 1.2 rad，约 69°。
//   - 形状扩大时，边缘带宽度为 7 的白色轮廓。
// 适用场景: 适合从画面中的物体发起转场，例如太阳、标志或按钮。
// 标签: mask, transition, iris, wipe, shape-wipe, reveal, scene-change, 遮罩, 转场, 圆形, 星形, 揭示, 场景
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/motion.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#017
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 17
// 17 マスクトランジション（丸や星の形の「窓」が広がり、その中に次の画面が見える）
CELLS[16] = abCell((k, w, h, u, a, b) => {
  const p = eIO(P(u, TA - 0.05, TB + 0.1));
  SCR[a](k, w, h);
  if (p > 0) {
    const c = a === 0 ? SUN : MOON, shape = () => { if (a === 0) { k.beginPath(); k.arc(c[0], c[1], lerp(0, 470, p), 0, 7); } else star(k, c[0], c[1], lerp(0, 1150, p), p * 1.2, 0.5); };
    k.save(); shape(); k.clip(); SCR[b](k, w, h); k.restore();
    if (p < 1) { k.strokeStyle = '#fff'; k.lineWidth = 7; k.lineJoin = 'round'; shape(); k.stroke(); }
  }
});
