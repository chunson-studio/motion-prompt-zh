// 涟漪效果 / Ripple effect
// 分类: lp (落地页与网页效果) · element 13 · 子类别: 悬停与点击交互
// 示例表现: 每次点击都会从点击位置向外扩散一个白色圆形，铺满按钮后逐渐消失。
// 动画参数与实现规则:
//   - 按钮尺寸为 280 × 56，橙色，圆角半径 10。在 0.7 s 点击中心左侧，在 2.9 s 点击中心右侧。
//   - 圆形以点击位置为中心，半径在 0.6 s 内从 0 增大到点击位置至按钮最远角的距离，使用三次缓出。
//   - 圆形为白色，初始不透明度 0.4；在点击后的 0.3 至 0.8 s 之间淡出到 0。
//   - 圆形被限制在按钮内，即 overflow: hidden；按钮文字显示在圆形上方。
// 适用场景: 适合在按钮或列表项的点击位置提供即时反馈。
// 标签: ripple, ink-ripple, material, click, tap, feedback, button, 涟漪, 点击, 按钮, 波纹, 扩散, 反馈
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#085
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 13
// 13 リップルエフェクト（押した場所から、ボタンの中に波紋が広がる）
CELLS[12] = (k, w, h, u) => {
  frame(k);
  const B = { x: 115, y: 78, w: 280, h: 56 }, TAPS = [[0.7, 165, 110], [2.9, 345, 100]], DUR = 0.6;
  const [cx, cy] = path([[0, 430, 160], [0.55, 165, 110], [1.6, 165, 110], [2.7, 345, 100], [3.6, 345, 100], [4.4, 430, 160], [4.8, 430, 160]], u);
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    txt(k, '立即申请', VW / 2, 38, `800 18px ${JP}`, C.ink, 'center'); bars(k, VW / 2 - 110, 50, 220, 1, C.faint, 6, 11, 1);
    k.save(); k.shadowColor = 'rgba(28,35,51,.18)'; k.shadowBlur = 8; k.shadowOffsetY = 3; k.fillStyle = C.acc; rr(k, B.x, B.y, B.w, B.h, 10); k.fill(); k.restore();
    k.save(); rr(k, B.x, B.y, B.w, B.h, 10); k.clip();
    for (const [tc, px, py] of TAPS) {
      const d = u - tc; if (d < 0 || d > 0.8) continue;
      const Rm = Math.max(Math.hypot(px - B.x, py - B.y), Math.hypot(px - B.x - B.w, py - B.y), Math.hypot(px - B.x, py - B.y - B.h), Math.hypot(px - B.x - B.w, py - B.y - B.h));
      k.fillStyle = `rgba(255,255,255,${0.4 * (1 - P(d, 0.3, 0.8))})`; k.beginPath(); k.arc(px, py, Rm * eOut(P(d, 0, DUR)), 0, 7); k.fill();
    }
    k.restore();
    txt(k, '发送', VW / 2, B.y + B.h / 2 + 1, `800 17px ${JP}`, '#fff', 'center', 'middle');
    click(k, cx, cy, u, TAPS.map(a => a[0]));
  });
};
