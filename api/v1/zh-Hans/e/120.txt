// 双重曝光 / Double exposure
// 分类: photo (摄影效果、光线与风格) · element 24 · 子类别: 色调与后期风格
// 示例表现: 背景变白，山丘、房屋、树木和太阳组成的景观缩小后，仅显示在人物半身像的轮廓内部。
// 动画参数与实现规则:
//   - 场景 3：左侧为半身像，头部圆心 x 140、y 82，半径 28 px；右侧为山丘、房屋、树木和太阳，太阳位于 x 432、y 54，半径 15 px。
//   - 背景：半身像轮廓外为纯白色 #FFFFFF。
//   - 半身像内部：同一景观缩放为 0.7×，调整位置，使太阳位于头部右上方，山脊横跨颈部。
//   - 内部颜色：天空 #D3DBE6，山丘 #9AA8BD 与 #76869F，地面 #55647E，树木 #3F4C66。
//   - 过渡：按照效果强度，将双重曝光画面叠加在原画面上。
//   - 时序：效果在 0.4 s 至 1.4 s 进入，保持到 4.0 s，并在 4.8 s 前完全消失；单次周期 = 5 s，使用三次缓入缓出，约等于 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合将人物与地点、记忆或内心世界联系起来的海报及封面。
// 标签: double exposure, multiple exposure, blend, silhouette, surreal, composite, 双重曝光, 人物轮廓, 叠加, 风景, 海报, 封面
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#120
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 24
// 二重露光: 背景が白く飛び、胸像の形の中にだけ丘と家と木と太陽が写り込む
CELLS[23] = cell((k, u) => {
  const a = env(u);
  const DX = Object.assign({}, BASE, { sky: '#D3DBE6', sky2: '#D3DBE6', far: '#9AA8BD', mid: '#76869F', ground: '#55647E', bld: '#E8ECF2', edge: '#55647E', win: '#9AA8BD', winOff: '#9AA8BD', tree: '#3F4C66', trunk: '#3F4C66' });
  const draw = kk => {
    kk.fillStyle = '#FFFFFF'; kk.fillRect(0, 0, CW, CH);
    kk.save(); bustPath(kk, QX, QY, QR); kk.clip();
    // 風景を0.7倍にして、太陽が頭の右上、丘の稜線が首のあたりに来るように置く
    kk.translate(QX, QY); kk.scale(0.7, 0.7); kk.translate(-415, -77);
    kk.fillStyle = DX.sky; kk.fillRect(0, -200, CW * 2, CH * 3);
    stageLand(kk, DX, { person: false });
    kk.fillStyle = DX.ground; kk.fillRect(-CW, 200, CW * 3, CH * 3);
    kk.restore();
  };
  if (a >= 0.999) { draw(k); return; }   // 効果が入りきったら直接描く（拡大表示でも輪郭がくっきり）
  stageLand(k, BASE);
  if (a < 0.002) return;
  const [L, kl] = layer('dx'); draw(kl);   // 入り・戻りの途中は、別の面に描いて重ねる
  k.save(); k.globalAlpha = a; k.drawImage(L, 0, 0); k.restore();
});
