// 伦勃朗光 / Rembrandt lighting
// 分类: photo (摄影效果、光线与风格) · element 05 · 子类别: 布光与时段
// 示例表现: 位于侧面 45°、上方 45° 的光源亮起；面部右半侧进入阴影，只有阴影一侧的脸颊保留一小块倒三角亮部。
// 动画参数与实现规则:
//   - 场景 1：人物半身像位于 y 150 的地平线前方，头部圆心为 x 270、y 78，半径 29 px，不绘制面部或头发。背景变暗，天空颜色为 #E7E5E1。
//   - 头部：x = 270 + 0.04·29 右侧为 #33405A 阴影，左侧为 #FFC75A 亮部。
//   - 光照三角形的顶点为 (x 270+0.24R, y 78−0.12R)、(x 270+0.7R, y 78−0.12R)、(x 270+0.24R, y 78+0.5R)，其中 R = 29；约宽 13 px、高 18 px，与明亮一侧之间隔着约 6 px 宽的阴影带。
//   - 躯干：中心左侧 −0.45×56 px 之外为亮部，右侧 +0.3×56 px 之外为阴影；布光图显示光源位于 45°，右上角读数为“45° · ↑45°”。
//   - 时序：效果在 0.4 s 至 1.4 s 进入，保持到 4.0 s，并在 4.8 s 前完全消失；单次周期 = 5 s，使用三次缓入缓出，约等于 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合带绘画感、雕塑感和古典气质的人像。
// 标签: rembrandt lighting, portrait lighting, triangle, 45 degree light, chiaroscuro, painterly, 伦勃朗光, 三角光, 人像, 古典, 侧上光, 布光
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#101
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 05
// レンブラントライティング: 斜め45°・上からの光。影側の頬に三角形の光が残る
CELLS[4] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a);
  bust(k, PX, PY, R, BASE.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
    kk.fillStyle = sh; kk.fillRect(PX + 0.04 * R, PY - R - 2, R * 2, R * 2 + 4);       // 鼻筋から右は影
    kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.clip();
    kk.fillStyle = lit; kk.beginPath();                                                   // 影側の頬に残る光の三角形
    kk.moveTo(PX + 0.24 * R, PY - 0.12 * R); kk.lineTo(PX + 0.7 * R, PY - 0.12 * R); kk.lineTo(PX + 0.24 * R, PY + 0.5 * R); kk.closePath(); kk.fill();
  });
  bust(k, PX, PY, R, 'rgba(0,0,0,0)', kk => {
    kk.fillStyle = 'rgba(0,0,0,0)';
    torso(kk, PX, PY, R, PX - W, PX - 0.45 * W, lit);
    torso(kk, PX, PY, R, PX + 0.3 * W, PX + W, sh);
  });
  plan(k, [[Math.PI / 4, a]]);
  hud(k, '45° · ↑45°');
});
