// 分页 / Pagination
// 分类: ui (UI 组件) · element 16 · 子类别: 切换与输入
// 示例表现: 依次点击页码 2 和 3，蓝色圆形选中标记移动，上方列表切换，最后返回第 1 页。
// 动画参数与实现规则:
//   - 在 542×191 的卡片上，周期为 4.5 s。页码“‹ 1 2 3 4 5 ›”位于 y = 160 并居中，每项直径 36，间隔 8 px；半径 18 的蓝色圆形标记当前页，内部文字为白色。
//   - 光标在 1.0 s 点击 2，在 2.1 s 点击 3，在 3.3 s 点击 1。
//   - 蓝色圆形每跨一页移动 44 px，每次持续 0.2 s，使用 easeInOutCubic（三次缓入缓出）；从 3 返回 1 时，一次移动 88 px。
//   - 在 1.05、2.15 和 3.35 s 前后各 0.2 s 范围内，上方三行列表的不透明度从 1→0→1，在最低点时更换内容。
// 适用场景: 适合将搜索结果、文章等长列表拆分为带页码的多页内容。
// 标签: pagination, pager, page-numbers, paging, next-prev, navigation, 分页, 页码, 列表, 切换, 导航, 搜索结果
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#064
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 16
// 16 ページネーション（Pagination）
CELLS[15] = ui((w, h, t) => {
  const page = t < 1.05 ? 0 : t < 2.15 ? 1 : t < 3.35 ? 2 : 0;
  const fade = Math.min(...[1.05, 2.15, 3.35].map(c => Math.abs(t - c) < .2 ? Math.abs(t - c) / .2 : 1));
  X.save(); X.globalAlpha = fade;
  for (let k = 0; k < 3; k++) { const y = 10 + k * 40; fillRR(24, y, 52, 32, 6, (k + page) % 3 === 0 ? C.mb : C.line); bar(90, y + 4, [200, 160, 240, 180][(k + page) % 4], C.line2, 10); bar(90, y + 20, [320, 280, 360][(k + 2 * page) % 3], C.line, 7); }
  X.restore();
  const items = ['‹', '1', '2', '3', '4', '5', '›'], d = 36, g = 8, total = items.length * d + (items.length - 1) * g, x0 = (w - total) / 2;
  const pos = eio(seg(t, 1.0, 1.2)) + eio(seg(t, 2.1, 2.3)) - 2 * eio(seg(t, 3.3, 3.5));
  circle(x0 + (1 + pos) * (d + g) + d / 2, 160, d / 2, C.blue);
  items.forEach((s, i) => { const cx = x0 + i * (d + g) + d / 2; const on = i >= 1 && i <= 5 && Math.abs(pos - (i - 1)) < .5; text(s, cx, 161, i === 0 || i === 6 ? 22 : 16, on ? '#fff' : '#666', 700, 'center', NUMF); });
  const px = i => x0 + i * (d + g) + d / 2 + 4;
  pointer([[.3, w + 30, h + 30], [.95, px(2), 166], [1.6, px(2), 166], [2.05, px(3), 166], [2.8, px(3), 166], [3.25, px(1), 166], [3.65, px(1), 166], [4.25, w + 30, h + 30]], [1.0, 2.1, 3.3], t);
});
