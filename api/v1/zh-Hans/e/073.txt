// 视差滚动 / Parallax scrolling
// 分类: lp (落地页与网页效果) · element 01 · 子类别: 滚动效果
// 示例表现: 页面向下滚动时，主标题以正常速度上移，天空与太阳、远山和近山则以不同的较慢速度移动。
// 动画参数与实现规则:
//   - 浏览器窗口的可视区域为 511 × 168。滚动位置在 0.3–1.9 s 从 0→120 px，在 2.9–4.3 s 从 120→0，使用三次缓入缓出。
//   - 各层移动量为滚动距离 × 系数：天空与太阳为 0.15，远山为 0.4，近山为 0.7。
//   - 主标题字号 28 px；标题、按钮与下一个区块均按 1.0 倍速移动，即正常滚动。
//   - 下一个白色区块从首屏主视觉底部的 168 px 处开始，并滑过移动较慢的背景。
//   - 右侧滚动条滑块随页面滚动而移动。
// 适用场景: 适合为首屏主视觉或区块图片增加空间纵深。
// 标签: parallax, parallax-scrolling, scroll, depth, layers, hero, background, 视差滚动, 多层背景, 景深, 首屏, 网页, 滚动
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#073
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 01
// 01 パララックススクロール（スクロールすると、空・山・見出しが違う速さで動く）
CELLS[0] = (k, w, h, u) => {
  frame(k);
  const s = steps([[0.3, 0], [1.9, 1], [2.9, 1], [4.3, 0]], u), y = s * 120;
  view(k, () => {
    const g = k.createLinearGradient(0, 0, 0, 200); g.addColorStop(0, '#9DB6FB'); g.addColorStop(1, '#E6ECFF');
    k.save(); k.translate(0, -y * 0.15);                         // 空と太陽: 0.15倍
    k.fillStyle = g; k.fillRect(0, 0, VW, 260); k.fillStyle = C.acc; k.beginPath(); k.arc(392, 54, 25, 0, 7); k.fill(); k.restore();
    k.save(); k.translate(0, -y * 0.4); mount(k, VW, 190, M1, '#7D9AF4', 0); k.restore();   // 遠い山: 0.4倍
    k.save(); k.translate(0, -y * 0.7); mount(k, VW, 205, M2, '#4B6BDB', 1); k.restore();   // 近い山: 0.7倍
    k.save(); k.translate(0, -y);                                // 見出し・ボタン・次のセクション: 1倍
    txt(k, '向山野，出发。', 34, 64, `800 28px ${JP}`, '#fff', 'left', 'alphabetic', '1px');
    bars(k, 34, 78, 160, 2, 'rgba(255,255,255,.85)');
    btn(k, 34, 106, 92, 26, '立即预订');
    k.fillStyle = '#fff'; k.fillRect(0, VH, VW, 200);
    txt(k, '热门路线', 34, VH + 34, `800 16px ${JP}`, C.ink);
    for (let i = 0; i < 3; i++) { pic(k, 34 + i * 152, VH + 50, 136, 58, i, 6); bars(k, 34 + i * 152, VH + 116, 110, 1, C.faint); }
    k.restore();
    sbar(k, s * 0.35, 0.4);
  });
};
