// 向下滚动提示 / Scroll-down cue
// 分类: lp2 (落地页与网页效果（第 2 弹）) · element 23 · 子类别: 亮点展示
// 示例表现: 首屏下方的“SCROLL”、鼠标轮廓内的小圆点，以及两侧箭头，持续重复向下运动。
// 动画参数与实现规则:
//   - 视口为 511 × 168。首屏下方居中显示“SCROLL”，字号 11 px、字距 3 px，以及鼠标轮廓，尺寸 24 × 38、圆角半径 12、线宽 2.2。每个节拍为 1.57 s，即每 4.7 s 三次、每个 9.4 s 的效果包周期六次。
//   - 内部橙色标记尺寸为 5 × 9，在节拍的前 70% 内下移 14 px，使用二次缓入缓出；在 45% 至 75% 之间渐出，下一个节拍时在顶部重新出现。
//   - 左右各相距 40 px 的位置，各有两个向下的 V 形箭头，线宽 2.5，彼此相隔半个节拍，向下漂移 24 px；不透明度按正弦变化，从 0→0.8→0。
// 适用场景: 适合占满屏幕的首屏，提示用户页面下方还有内容。
// 标签: scroll-cue, scroll-indicator, scroll-down, hero, arrow, loop, 滚动提示, 向下, 箭头, 首屏, 引导, 循环
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit lp2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp2.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#167
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 23
// 23 スクロールダウンの合図（ファーストビューの下で「SCROLL」とマウスの形の中の点が、下へ促す動きをくり返す）
CELLS[22] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const T = CYC / 3, f = mod(u, T) / T;
  view(k, () => {
    k.fillStyle = '#FFF4EA'; k.fillRect(0, 0, VW, VH);
    txt(k, '欢迎来到，', 40, 40, `800 22px ${JP}`, C.ink); txt(k, '山顶旅宿。', 40, 70, `800 22px ${JP}`, C.ink);
    bars(k, 40, 84, 170, 2, '#EAD9CB');
    pic(k, 286, 10, 200, 92, 1, 10);
    // 合図
    const mx = VW / 2;
    txt(k, 'SCROLL', mx, 116, `800 11px ${EN}`, C.ink, 'center', 'middle', '3px');
    k.strokeStyle = C.ink; k.lineWidth = 2.2; rr(k, mx - 12, 126, 24, 38, 12); k.stroke();
    const dy = eIOs(clamp(f / 0.7)) * 14, da = 1 - clamp((f - 0.45) / 0.3);
    k.fillStyle = `rgba(255,90,44,${da})`; rr(k, mx - 2.5, 133 + dy, 5, 9, 2.5); k.fill();
    // 両わきの矢印も下へ流れる
    [mx - 40, mx + 40].forEach(x => { for (let j = 0; j < 2; j++) { const ff = mod(f + j * 0.5, 1), yy = 128 + ff * 24, al = Math.sin(Math.PI * ff); k.strokeStyle = `rgba(28,35,51,${0.8 * al})`; k.lineWidth = 2.5; k.lineCap = 'round'; k.lineJoin = 'round'; k.beginPath(); k.moveTo(x - 8, yy); k.lineTo(x, yy + 8); k.lineTo(x + 8, yy); k.stroke(); } });
  });
};
