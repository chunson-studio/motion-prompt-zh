// 轮播图 / Carousel
// 分类: ui (UI 组件) · element 10 · 子类别: 切换与输入
// 示例表现: 每次点击右箭头，三张不同图片组成的横排向左移动一格，下方指示点同步变化。
// 动画参数与实现规则:
//   - 在 542×191 的卡片上，周期为 4.5 s。幻灯片尺寸 330×138、圆角半径 8，居中显示，间距 18 px；两侧边缘露出相邻图片的一部分。
//   - 左右两侧为带箭头的白色圆形按钮，半径 18、阴影模糊半径 10；下方有三个圆点，当前项为蓝色 16×8，其余为 8×8。
//   - 光标在 1.0、2.1 和 3.2 s 点击位置 (514, 88) 的右箭头。
//   - 每次点击使图片横排在 0.4 s 内向左移动 348 px，即一张图片加间距，使用 easeInOutCubic（三次缓入缓出）。
//   - 三张图片分别为风景、三个圆形和柱状图，循环排列，因此移动三次后返回第一张。
// 适用场景: 适合在有限空间中逐一展示多张图片或推荐内容。
// 标签: carousel, slider, slideshow, gallery, swipe, paging-dots, 轮播图, 图片, 滑动, 指示点, 循环, 展示
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#058
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 10
// 10 カルーセル（Carousel）
CELLS[9] = ui((w, h, t) => {
  const p = eio(seg(t, 1.0, 1.4)) + eio(seg(t, 2.1, 2.5)) + eio(seg(t, 3.2, 3.6));
  const sw = 330, gap = 18, cx = w / 2;
  X.save(); X.beginPath(); X.rect(0, 0, w, 160); X.clip();
  for (let j = Math.floor(p) - 2; j <= Math.floor(p) + 2; j++) {
    const x = cx - sw / 2 + (j - p) * (sw + gap), idx = ((j % 3) + 3) % 3;
    if (idx === 0) mountains(x, 14, sw, 138);
    if (idx === 1) { fillRR(x, 14, sw, 138, 8, '#e3ebfd'); circle(x + 90, 83, 34, C.mb); circle(x + 165, 83, 34, C.sb); circle(x + 240, 83, 34, C.mb); }
    if (idx === 2) { fillRR(x, 14, sw, 138, 8, C.lb); for (let k = 0; k < 5; k++) { const bh = [40, 70, 55, 95, 80][k]; fillRR(x + 60 + k * 46, 132 - bh, 30, bh, 4, k === 3 ? C.blue : C.mb); } }
  }
  X.restore();
  [[30, 83, Math.PI / 2], [w - 30, 83, -Math.PI / 2]].forEach(([x, y, r]) => { shadowed(() => circle(x, y, 18, '#fff'), 10, 3, .2); chevron(x, y, 5, r, '#555'); });
  const act = Math.round(p) % 3;
  for (let k = 0; k < 3; k++) { const on = k === act; fillRR(w / 2 - 30 + k * 22 - (on ? 4 : 0), 172, on ? 16 : 8, 8, 4, on ? C.blue : C.line2); }
  pointer([[.3, w + 30, h + 30], [.95, w - 28, 88], [3.4, w - 28, 88], [4.2, w + 30, h + 30]], [1.0, 2.1, 3.2], t);
});
