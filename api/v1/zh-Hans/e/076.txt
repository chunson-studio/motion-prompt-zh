// 滚动固定 / Pinning
// 分类: lp (落地页与网页效果) · element 04 · 子类别: 滚动效果
// 示例表现: 深蓝色区块占满视口后固定不动；继续滚动时，数字与图片依次切换为 1→2→3，随后区块恢复滚动。
// 动画参数与实现规则:
//   - 滚动位置在 0.3–3.0 s 从 0→500 px，在 3.5–4.5 s 从 500→0，使用三次缓入缓出。
//   - 深蓝色区块高 168，在滚动距离达到 110 px 时到达顶边，并在接下来的 300 px 滚动区间内保持固定，类似 GSAP ScrollTrigger 的 pin 效果。
//   - 固定期间的进度为 p = (scroll − 110) / 300。左侧长 124 的竖线随 p 逐渐填充为橙色。
//   - 在 p = 1/3 和 2/3 附近切换到下一步骤，过渡区间为 ±0.07，使用三次缓入缓出；字号 46 px 的数字上滚 52 px，尺寸 226 × 124 的图片上滑 124 px。
//   - 经过 300 px 后解除固定，当前区块连同下方白色区块一起向上滚动。
// 适用场景: 适合在一屏内介绍操作步骤、产品功能或前后对比。
// 标签: pin, pinning, scrolltrigger, gsap, scroll-driven, steps, storytelling, 滚动固定, 钉住, 步骤展示, 进度, 图片切换, 滚动动画
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#076
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 04
// 04 ピン留め（セクションが画面に止まったまま、中身だけがスクロールで進む）
CELLS[3] = (k, w, h, u) => {
  frame(k);
  const y = steps([[0.3, 0], [3.0, 500], [3.5, 500], [4.5, 0]], u);
  const PIN0 = 110, PD = 300;
  const top = y < PIN0 ? PIN0 - y : y < PIN0 + PD ? 0 : PIN0 + PD - y, p = clamp((y - PIN0) / PD);
  // 中身は 1/3・2/3 の所で次のステップへ（前後 ±0.07 の範囲で easeInOut）
  const f = eIO(P(p, 1 / 3 - 0.07, 1 / 3 + 0.07)) + eIO(P(p, 2 / 3 - 0.07, 2 / 3 + 0.07));
  view(k, () => {
    txt(k, '3步轻松完成', 34, 56 - y, `800 22px ${JP}`, C.ink); bars(k, 34, 72 - y, 240, 2);
    k.save(); k.translate(0, top);
    k.fillStyle = C.ink; k.fillRect(0, 0, VW, VH);
    // 左: STEP の番号（スクロール量に合わせて回る）
    txt(k, 'STEP', 34, 40, `700 12px ${EN}`, 'rgba(255,255,255,.6)', 'left', 'alphabetic', '2px');
    k.save(); k.beginPath(); k.rect(30, 46, 120, 52); k.clip();
    for (let i = 0; i < 3; i++) txt(k, '0' + (i + 1), 32, 90 + (i - f) * 52, `800 46px ${EN}`, '#fff');
    k.restore();
    bars(k, 34, 110, 150, 3, 'rgba(255,255,255,.28)');
    // 進み具合の線
    k.fillStyle = 'rgba(255,255,255,.16)'; rr(k, 232, 22, 4, 124, 2); k.fill();
    k.fillStyle = C.acc; rr(k, 232, 22, 4, 124 * p, 2); k.fill();
    // 右: 写真が順に入れ替わる
    k.save(); rr(k, 256, 22, 226, 124, 8); k.clip();
    for (let i = 0; i < 3; i++) pic(k, 256, 22 + (i - f) * 124, 226, 124, i, 0);
    k.restore();
    k.fillStyle = '#fff'; k.fillRect(0, VH, VW, 200); bars(k, 34, VH + 30, 300, 3);
    k.restore();
    sbar(k, y / 500, 0.3);
  });
};
