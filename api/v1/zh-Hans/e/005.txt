// 遮罩显现 / Mask reveal
// 分类: motion (动态图形) · element 05 · 子类别: 入场与文字
// 示例表现: 标题文字从一条线所标示的不可见边缘后方滑出。
// 动画参数与实现规则:
//   - 每 2.35 s 完成一轮。珊瑚色分隔线尺寸为 380 × 6，位于 y = 124；在 0.1–0.5 s 从中央向两侧伸展，使用三次缓入缓出，在 2.05–2.3 s 缩回消失。
//   - 上方文字为 98 px 白色粗体，裁切在分隔线上方区域，y 范围为 8–119；在 0.35–1.0 s 从下方 120 px 处向上移入原位，使用五次缓出。
//   - 下方文字为 32 px 黄色，裁切在分隔线下方区域，y 范围为 129–181；在 0.55–1.15 s 从上方 60 px 处向下移入原位，使用五次缓出。
//   - 退场：上方文字在 1.7–2.05 s 向上移动 150 px，使用 backIn，s = 1.2，先略微反向下沉；下方文字在 1.75–2.05 s 向下移动 70 px，使用三次缓入。
//   - 不透明度始终不变，文字仅通过离开裁切区域而隐藏。
// 适用场景: 适合希望以干净、有秩序的方式出现的标题和小标题。
// 标签: mask, reveal, clip, wipe, title, text, slide-up, headline, 遮罩, 显现, 标题, 裁切, 文字, 入场
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/motion.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#005
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 05
// 05 マスクリビール（見えない枠＝線の向こうから、見出しがせり上がる）
CELLS[4] = (k, w, h, u) => {
  stage(k);
  const T = CYC / 2, lu = mod(u, T), X = w / 2, LY = 124;
  const lg = eIO(P(lu, 0.1, 0.5)) * (1 - eIO(P(lu, 2.05, 2.3)));
  // 上の段：線の下から上へ
  const up = (1 - eOut5(P(lu, 0.35, 1.0))) * 120 + backIn(P(lu, 1.7, 2.05), 1.2) * -150;
  k.save(); k.beginPath(); k.rect(0, 8, w, LY - 8 - 5); k.clip();
  txt(k, 'Hello!', X, LY - 14 + up, `800 98px ${DISP}`, '#fff', 'center', 'alphabetic', '1px');
  k.restore();
  // 下の段：線の上から下へ
  const dn = (1 - eOut5(P(lu, 0.55, 1.15))) * -60 + eIn(P(lu, 1.75, 2.05)) * 70;
  k.save(); k.beginPath(); k.rect(0, LY + 5, w, 52); k.clip();
  txt(k, '很高兴认识你', X, LY + 44 + dn, `700 32px ${JP}`, C.yel, 'center', 'alphabetic', '6px');
  k.restore();
  if (lg > 0.005) { k.fillStyle = C.coral; rr(k, X - 190 * lg, LY - 3, 380 * lg, 6, 3); k.fill(); }
};
