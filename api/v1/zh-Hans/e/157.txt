// 悬停图片预览 / Hover image reveal
// 分类: lp2 (落地页与网页效果（第 2 弹）) · element 13 · 子类别: 鼠标与输入交互
// 示例表现: 鼠标悬停在纯文字列表的某一行时，该行对应的照片会在光标旁渐入，并跟随光标移动。
// 动画参数与实现规则:
//   - 视口为 511 × 168，显示四行纯文字列表，行高 39 px，名称字号 17 px。光标沿各行向下移动，每行停留约 1 s。一次演示为 4.7 s，每个 9.4 s 的效果包周期内执行两次。
//   - 被悬停的行在 0.25 s 内使用三次缓出：名称向右移动 8 px，名称与最右侧箭头变为橙色；离开时在 0.25 s 内恢复。
//   - 照片使用该行对应的图像，尺寸 132 × 84、圆角半径 6，带投影；位于光标右侧 26 px、略高于光标的位置，并限制其垂直范围，避免超出窗口。
//   - 同样在 0.25 s 内，照片不透明度从 0 变为 1，缩放比例从 0.85× 变为 1×。照片无延迟地跟随光标；移到下一行时，旧照片淡出的同时新照片淡入。
// 适用场景: 适合作品、项目或房产列表，保持纯文字布局的轻巧，同时让用户预览感兴趣条目的照片。
// 标签: hover, image-reveal, list, cursor-follow, preview, portfolio, 悬停, 图片预览, 列表, 跟随, 作品集, 光标
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit lp2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp2.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#157
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 13
// 13 ホバーで画像プレビュー（文字だけのリストの行にカーソルを乗せると、その行の写真がカーソルの横に出る）
CELLS[12] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const NAMES = ['极光山丘', '湖畔小屋', '海边别墅', '星空露台'], RT = 6, RH = 39;
  const cur = t => path([[0, 140, 200], [0.4, 150, 26], [1.15, 190, 28], [1.4, 205, 65], [2.15, 230, 67], [2.4, 245, 104], [3.15, 265, 106], [3.4, 280, 143], [3.9, 292, 144], [4.3, 150, 200], [4.7, 140, 200]], t);
  const [cx, cy] = cur(u);
  view(k, () => {
    NAMES.forEach((s, i) => {
      const top = RT + i * RH, inside = t => { const [a, b] = cur(t); return b > top && b < top + RH && a > 20 && a < 490; };
      const q = trans(inside, u, 0.25);
      txt(k, '0' + (i + 1), 30, top + 25, `800 12px ${EN}`, C.sub);
      txt(k, s, 64 + 8 * q, top + 26, `800 17px ${JP}`, mix(C.ink, C.acc, q));
      txt(k, '→', 476, top + 25, `700 15px ${EN}`, mix('#B7BFCC', C.acc, q), 'right');
      k.fillStyle = C.rule; k.fillRect(24, top + RH, VW - 48, 1);
    });
    NAMES.forEach((s, i) => {
      const top = RT + i * RH, q = trans(t => { const [a, b] = cur(t); return b > top && b < top + RH && a > 20 && a < 490; }, u, 0.25);
      if (q <= 0) return;
      const iw = 132, ih = 84, px = cx + 26, py = clamp(cy - 46, 4, VH - ih - 4), sc = 0.85 + 0.15 * q;
      k.save(); k.globalAlpha = q; k.translate(px + iw / 2, py + ih / 2); k.scale(sc, sc);
      k.shadowColor = 'rgba(28,35,51,.25)'; k.shadowBlur = 12; k.shadowOffsetY = 4; k.fillStyle = '#fff'; rr(k, -iw / 2, -ih / 2, iw, ih, 6); k.fill(); k.shadowColor = 'transparent';
      pic(k, -iw / 2, -ih / 2, iw, ih, [0, 2, 1, 2][i], 6);
      k.restore();
    });
    cursor(k, cx, cy);
  });
};
