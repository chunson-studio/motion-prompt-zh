// 子弹时间 / Bullet time
// 分类: camera2 (镜头运动与转场（第 2 弹）) · element 23 · 子类别: 转场与剪辑
// 示例表现: 人物跳至最高点附近时，时间几乎停止，相机则绕着人物移动。
// 动画参数与实现规则:
//   - 影像时间：0–1.5 s 正常流逝，1.5–3.3 s 以 0.1× 速度流逝，即 1.8 s 内仅经过 0.18 s 的影像时间，之后恢复正常速度；恢复后影像时间 = u - 1.62。
//   - 主体：前方 9 m 处的人物，在影像时间 0.6–2.6 s 沿抛物线跳跃，最高 1.1 m，双臂举起。慢速阶段停留在最高点附近，即跳跃进程的 45–54%；九片落叶按同一影像时间飘落。
//   - 相机：慢速阶段沿以人物为中心、半径 6 m 的圆弧移动，角度从 -0.6 rad 到 +0.6 rad，约 69°，使用三次缓入缓出，约等于 cubic-bezier(0.65,0,0.35,1)。始终朝向圆心，高度 1.6 m、俯仰角 -0.05 rad。
//   - 接下来的 4.7 s 演示反向绕行，从 +0.6→-0.6 rad，因此完整的 9.4 s 循环无缝衔接；#030 环绕镜头并不会让时间停止。
//   - 地面上有半径 1.3 m 的虚线圆，10 m 处环绕一圈树木。右上角读数依次为“×1.0”→“BULLET ×0.1”→“×1.0”。
// 适用场景: 适合从多个角度呈现决定性瞬间，或作为动作场面的高潮。
// 标签: bullet time, time slice, frozen moment, orbit, slow motion, matrix, effect, 子弹时间, 慢动作, 环绕, 跳跃, 时间, 动作
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit camera2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera2.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#191
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 23
// 23 バレットタイム: 跳躍の頂点で時間がほぼ止まり、その間にカメラが人物の周りを回り込む（#030 オービットは時間が止まらない）
CELLS[22] = shot((k, w, h, u, cyc) => {
  const T0 = 1.5, T1 = 3.3, SLOW = 0.1;
  const tj = u < T0 ? u : u < T1 ? T0 + (u - T0) * SLOW : u - (T1 - T0) * (1 - SLOW);   // 実時間→映像時間
  const q = P(tj, 0.6, 2.6), e = eIO(P(u, T0, T1));
  const a = cyc % 2 === 0 ? lerp(-0.6, 0.6, e) : lerp(0.6, -0.6, e), R = 6, P0 = [0, 0, 9];
  const c = cam({ x: P0[0] + Math.sin(a) * R, z: P0[2] - Math.cos(a) * R, yaw: -a, y: 1.6, pitch: -0.05 });
  world(k, w, h, c, {
    person: { z: 9, y: jumpY(q), armUp: q > 0 && q < 1, face: a }, ring: [0, 9, 10], trees: false, extra: leaves(tj, 1.5).map(l => [l[0], l[1], l[2] + 3, l[3]]),
    after: (kk, cc) => { kk.strokeStyle = C.blueM; kk.lineWidth = 2; kk.setLineDash([5, 5]); kk.beginPath(); for (let i = 0; i <= 48; i++) { const b = i / 48 * Math.PI * 2; const qq = proj(cc, Math.sin(b) * 1.3, 0, 9 + Math.cos(b) * 1.3); i ? kk.lineTo(qq.x, qq.y) : kk.moveTo(qq.x, qq.y); } kk.stroke(); kk.setLineDash([]); }
  });
  return u >= T0 && u < T1 ? `BULLET ×${SLOW.toFixed(1)}` : '×1.0';
});
