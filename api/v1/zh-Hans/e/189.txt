// 切出镜头 / Cutaway
// 分类: camera2 (镜头运动与转场（第 2 弹）) · element 21 · 子类别: 转场与剪辑
// 示例表现: 从讲话者切到倾听的观众，再切回讲话者；插入画面展示的是另一个主体。
// 动画参数与实现规则:
//   - 衔接：两次直接剪切，过渡为 0 帧；在 1.7 s 切到另一个主体，在 3.0 s 返回原场景，离开原场景共 1.3 s。
//   - 主镜头 A：相机固定，高度 1.4 m；前方 5.5 m 处的人物讲话，四肢按 0.45·sin(u×6) rad 摆动，其右侧 0.9 m 有一座灰色讲台。
//   - 另一主体 CUTAWAY：倾听的观众，共五人，位于 3.1–4.6 m 远处，使用蓝色与浅蓝色，双臂举起，按 |sin(t×9+i)|×5 cm 上下弹动；没有背景，表示另一处空间。
//   - 底部时间条：灰色 A、蓝色另一主体、灰色 A，并带播放头；本组第 17 项插入镜头则展示同一场景的细节。
//   - 右上角读数依次为“SHOT A”→“CUTAWAY”→“SHOT A”；在 4.7 s 直接切回起始状态。
// 适用场景: 适合展示反应，或掩盖时间跳跃及删去口误后的剪辑点。
// 标签: cutaway, reaction shot, b-roll, cut away, audience, edit, 切出镜头, 剪辑, 反应, 观众, 切换, 衔接
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit camera2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera2.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#189
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 21
// 21 カットアウェイ: 話す人物から、別の対象（聞いている客席）へ切り替えて戻る。17 インサートは同じ場面の細部
const CUT = [1.7, 3.0];
CELLS[20] = shot((k, w, h, u) => {
  const away = u >= CUT[0] && u < CUT[1];
  if (!away) world(k, w, h, cam({ y: 1.4 }), { person: { x: 0, z: 5.5, walk: 0.45 * Math.sin(u * 6) }, extra: [[0.9, 0, 5.6, (kk) => { kk.fillStyle = C.g3; kk.fillRect(-0.3, 0, 0.6, 1.1); kk.fillRect(-0.4, 1.05, 0.8, 0.1); }]] });
  else {
    const t = u - CUT[0], ex = [];
    [[-1.6, 3.2], [-0.2, 3.6], [1.3, 3.1], [0.6, 4.4], [-1.1, 4.6]].forEach(([x, z], i) => ex.push([x, 0, z, (kk) => { kk.save(); kk.translate(0, Math.abs(Math.sin(t * 9 + i)) * 0.05); person(kk, 1, { armUp: true, col: i % 2 ? C.blue : C.blueM }); kk.restore(); }]));
    world(k, w, h, cam({ y: 1.6, pitch: -0.08 }), { buildings: false, trees: false, mountains: false, sun: null, extra: ex });
  }
  strip(k, w, h, [[0, CUT[0], 'A', C.g2], [CUT[0], CUT[1], '其他主体', C.blue], [CUT[1], CYC, 'A', C.g2]], u);
  return away ? 'CUTAWAY' : 'SHOT A';
});
