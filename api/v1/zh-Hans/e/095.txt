// 动态渐变 / Animated gradient
// 分类: lp (落地页与网页效果) · element 23 · 子类别: 加载、导航与循环动效
// 示例表现: 主标题背后，柔和的橙色、浅橙色、蓝色和浅蓝色色块沿圆周漂移，持续形成不同的混合色彩。
// 动画参数与实现规则:
//   - 底色为 #FFF1EB，其上叠加四个径向渐变，均从中心颜色过渡到边缘透明，不透明度各为 0.85。
//   - 各渐变分别为橙色（半径 230）、浅橙色 #FFB08F（210）、蓝色（240）和浅蓝色 #9DB6FB（200）。
//   - 各渐变中心每 4.8 s 绕椭圆运动一周；x 半径分别为 70、60、80、50，y 半径分别为 30、40、30、35，起始角度分别为 0、2.1、4.2、1.0 rad。
//   - 白色主标题字号 34 px，与按钮一起置于渐变上方；只有背景渐变移动。
// 适用场景: 适合无需照片、希望以动态色彩丰富首屏或标题背景的场景。
// 标签: gradient, animated-gradient, mesh-gradient, background, hero, ambient, loop, 动态渐变, 色块, 背景, 首屏, 椭圆运动, 颜色混合
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#095
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 23
// 23 グラデーションアニメーション（背景のグラデーションの色が、ゆっくり流れるように動き続ける）
CELLS[22] = (k, w, h, u) => {
  frame(k);
  const a = 2 * Math.PI * u / CYC;                        // 4.8秒で1周
  const BLOBS = [[C.acc, 150, 60, 230, 70, 30, 0], ['#FFB08F', 380, 40, 210, 60, 40, 2.1], [C.blue, 330, 150, 240, 80, 30, 4.2], ['#9DB6FB', 90, 170, 200, 50, 35, 1.0]];
  view(k, () => {
    k.fillStyle = '#FFF1EB'; k.fillRect(0, 0, VW, VH);
    for (const [col, x0, y0, r, ax, ay, ph] of BLOBS) {
      const x = x0 + ax * Math.cos(a + ph), y = y0 + ay * Math.sin(a + ph);
      const g = k.createRadialGradient(x, y, 0, x, y, r); g.addColorStop(0, col); g.addColorStop(1, 'rgba(255,255,255,0)');
      k.globalAlpha = 0.85; k.fillStyle = g; k.fillRect(0, 0, VW, VH); k.globalAlpha = 1;
    }
    txt(k, 'Hello, Future.', 40, 82, `800 34px ${EN}`, '#fff', 'left', 'alphabetic', '1px');
    bars(k, 40, 100, 200, 2, 'rgba(255,255,255,.8)', 6, 12, 0.6);
    btn(k, 40, 128, 104, 26, '开始使用', '#fff', C.ink);
  });
};
