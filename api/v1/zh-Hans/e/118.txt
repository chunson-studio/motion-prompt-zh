// 胶片颗粒 / Film grain
// 分类: photo (摄影效果、光线与风格) · element 22 · 子类别: 色调与后期风格
// 示例表现: 细小颗粒覆盖同一景观，整体颜色略微偏向温暖的奶油色。
// 动画参数与实现规则:
//   - 场景 3：左侧为半身像，头部圆心 x 140、y 82，半径 28 px；右侧为山丘、房屋、树木和太阳，太阳位于 x 432、y 54，半径 15 px。
//   - 颗粒为 2,600 个使用固定随机种子生成的方块，尺寸 0.8–1.7 px；其中 55% 为深色 #3A3530，45% 为白色。
//   - 颗粒不透明度为 16–38% × 效果强度；颗粒保持静止，不闪烁。
//   - 颜色：所有颜色都向 #F2DDBE 偏移 16%。
//   - 时序：效果在 0.4 s 至 1.4 s 进入，保持到 4.0 s，并在 4.8 s 前完全消失；单次周期 = 5 s，使用三次缓入缓出，约等于 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合削弱过于干净的数码观感，增加模拟胶片质感。
// 标签: film grain, grain, noise, analog, 35mm film, texture, warm tone, 胶片颗粒, 胶片感, 噪点, 复古, 纹理, 暖色
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#118
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 22
// フィルムグレイン: 細かい粒子が全体に乗り、全体が少し暖かい色に寄る
CELLS[21] = cell((k, u) => {
  const a = env(u);
  const FILM = mapPal(BASE, c => mixc(c, '#F2DDBE', 0.16));
  const p = pal(BASE, FILM, a);
  const G = once('grain', () => { const r = rng(22), o = []; for (let i = 0; i < 2600; i++) o.push([r() * CW, r() * CH, 0.8 + r() * 0.9, r() < 0.55 ? 0 : 1, 0.16 + r() * 0.22]); return o; });
  const grain = kk => { kk.save(); for (const [x, y, s, d, al] of G) { kk.globalAlpha = al * a; kk.fillStyle = d ? '#FFFFFF' : '#3A3530'; kk.fillRect(x, y, s, s); } kk.restore(); };
  stageLand(k, p); grain(k);
});
