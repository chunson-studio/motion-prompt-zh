// 步骤指示器 / Step indicator
// 分类: ui (UI 组件) · element 20 · 子类别: 导航与状态反馈
// 示例表现: 每次点击“下一步”，四个步骤圆点中的当前步骤会变成勾选状态，蓝色连接线向下一步骤延伸。
// 动画参数与实现规则:
//   - 卡片尺寸 542×191，周期 4.5 s。四个步骤为“填写、确认、支付、完成”，位置分别是 x 80、207、334、461，y 35；圆点半径 17，由粗 4 px 的灰线连接。
//   - 光标在 0.9、1.7 和 2.5 s 点击右下角的“下一步”按钮，按钮尺寸为 110×38。
//   - 每次点击，当前步骤变成蓝底白色对勾，下一个步骤成为当前步骤，显示蓝底白色编号；尚未到达的步骤是白底灰色边框。
//   - 每次点击后，蓝色连接线在 0.3 s 内延长 127 px，使用 easeInOutCubic；第三次点击后，四个步骤都显示对勾。
//   - 重置：步骤指示器在 3.75–4.05 s 淡出，在 4.05–4.3 s 从步骤 1 状态淡入。
// 适用场景: 适合注册、结账等多步骤流程，展示当前进度和剩余步骤。
// 标签: step-indicator, stepper, steps, progress-steps, wizard, multi-step, checkout, 步骤指示器, 进度, 注册, 结账, 多步骤, 连接线
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#068
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 20
// 20 ステップインジケーター（Step indicator）
CELLS[19] = ui((w, h, t) => {
  const clicks = [.9, 1.7, 2.5]; let step = 0; clicks.forEach(c => { if (t >= c) step++; });
  const reset = seg(t, 3.75, 4.05), back = seg(t, 4.05, 4.3);
  const showStep = reset >= 1 ? 0 : step;
  const xs = [80, 207, 334, 461], labs = ['填写', '确认', '支付', '完成'];
  const prog = clicks.reduce((s, c) => s + eio(seg(t, c, c + .3)), 0);
  X.save(); X.globalAlpha = reset < 1 ? 1 - reset : back;
  const pr = reset >= 1 ? 0 : prog;
  fillRR(xs[0], 33, xs[3] - xs[0], 4, 2, '#e3e1df'); fillRR(xs[0], 33, (xs[1] - xs[0]) * pr, 4, 2, C.blue);
  xs.forEach((x, i) => {
    const done = i < showStep || (showStep === 3 && i === 3), cur = i === showStep && showStep < 3;
    circle(x, 35, 17, done || cur ? C.blue : '#fff'); if (!(done || cur)) ring(x, 35, 16, C.line2, 2);
    if (done) check(x, 35, 7, '#fff', 2.6); else text(String(i + 1), x, 36, 16, cur ? '#fff' : '#9a9896', 700, 'center', NUMF);
    text(labs[i], x, 70, 14, done || cur ? C.blue : '#9a9896', 600, 'center');
  });
  X.restore();
  strokeRR(24, 96, 494, 30, 7, C.line2, 1.5); bar(36, 106, 160, C.line); bar(24, 148, 200, C.line);
  fillRR(w - 24 - 110, 142, 110, 38, 8, C.blue); text('下一步', w - 24 - 55, 161, 16, '#fff', 600, 'center');
  pointer([[.3, w + 30, h + 30], [.85, w - 74, 166], [2.6, w - 74, 166], [3.3, w + 30, h + 30]], clicks, t);
});
