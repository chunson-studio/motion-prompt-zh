// 追随拍摄 / Panning shot
// 分类: photo (摄影效果、光线与风格) · element 12 · 子类别: 镜头与相机效果
// 示例表现: 相机追随移动的汽车，汽车始终清晰地处在画面中央，只有后方建筑、树木和路灯形成横向拖影。
// 动画参数与实现规则:
//   - 街角中的建筑、树木和路灯位于 y 132–162 的道路后方，不绘制大型前景人物。
//   - 汽车在无效果时于 5 s 内从 x −80 行驶到 620；效果开启时，汽车保持在 x 270 ±10 px。
//   - 拖影：将背景在横向 46 px 范围内绘制 12 次并取平均；扩散范围从 0 → 46 px。
//   - 车轮：辐条线每 5 s 旋转 12 圈。
//   - 右上角快门速度从 1/1000 → 1/15。
//   - 时序：效果在 0.4 s 至 1.4 s 进入，保持到 4.0 s，并在 4.8 s 前完全消失；单次周期 = 5 s，使用三次缓入缓出，约等于 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合在静态照片中表现汽车、自行车或跑者的速度。
// 标签: panning, panning shot, motion blur, speed, sports photography, slow shutter, 追随拍摄, 摇拍, 速度感, 运动模糊, 汽车, 快门
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#108
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 12
// 流し撮り: 走る車を追ってカメラを振ると、車はくっきり、背景だけ横に流れる
CELLS[11] = cell((k, u) => {
  const a = env(u);
  const drive = lerp(-80, 620, u / CYC), x = lerp(drive, 270 + 10 * Math.sin(u / CYC * TAU), a);
  const S = 46 * a, n = S < 0.5 ? 1 : 12;
  for (let i = 0; i < n; i++) {   // 背景を横にずらして重ね、平均をとる（流れ）
    k.save(); k.globalAlpha = 1 / (i + 1); k.translate(n > 1 ? lerp(-S / 2, S / 2, i / (n - 1)) : 0, 0); stageStreet(k, BASE, { car: false, person: false }); k.restore();
  }
  const D = pen(k); car(D, x, BASE, { head: BASE.lit });
  // 車輪の回転（1周に12回転＝継ぎ目なし）
  k.save(); k.strokeStyle = BASE.tire; k.lineWidth = 1.6;
  for (const wx of [x - 21, x + 21]) { const t = u / CYC * TAU * 12; k.beginPath(); k.moveTo(wx - Math.cos(t) * 5, 156 - Math.sin(t) * 5); k.lineTo(wx + Math.cos(t) * 5, 156 + Math.sin(t) * 5); k.stroke(); }
  k.restore();
  hud(k, a > 0.5 ? '1/15' : '1/1000');
});
