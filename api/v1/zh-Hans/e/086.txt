// 下划线动画 / Underline animation
// 分类: lp (落地页与网页效果) · element 14 · 子类别: 悬停与点击交互
// 示例表现: 光标所在链接的下方，橙色下划线从左侧展开；光标离开时，下划线向右侧收走。
// 动画参数与实现规则:
//   - 三个链接使用 20 px 粗体，间距 56。光标从左向右，分别停留约 1 s；命中区域在文字两侧各增加 6 px。
//   - 下划线为橙色，粗 3，位于文字基线下方 8 px。
//   - 悬停时，线条宽度在 0.3 s 内从 0→100%，从左端展开，使用三次缓出。
//   - 离开时，线条左端在 0.3 s 内向右移动，使整条线向右退出，类似把 transform-origin 设在右侧后执行 scaleX。
// 适用场景: 适合菜单和文字链接，以轻量的方式提示光标位置。
// 标签: underline, hover, link, navigation, menu, text-link, scalex, 下划线动画, 文字链接, 悬停, 导航, 展开, 交互反馈
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#086
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 14
// 14 下線アニメーション（リンクにカーソルを乗せると下線が左から伸び、離れると右へ抜ける）
CELLS[13] = (k, w, h, u) => {
  frame(k);
  const LINKS = ['服务', '案例', '公司简介'], F = `800 20px ${JP}`, GAP = 56, LY = 84, DUR = 0.3;
  const cur = t => path([[0, 120, 150], [0.45, 170, 80], [1.3, 186, 84], [1.55, 268, 80], [2.4, 282, 84], [2.65, 372, 80], [3.5, 390, 84], [3.95, 430, 150], [4.8, 120, 150]], t);
  view(k, () => {
    logo(k, 22, 28); bars(k, 90, 132, 331, 2, C.faint, 6, 12, 0.6);
    k.font = F; const ws = LINKS.map(s => k.measureText(s).width); let x = (VW - ws.reduce((a, b) => a + b, 0) - GAP * (LINKS.length - 1)) / 2;
    LINKS.forEach((s, i) => {
      const x0 = x, x1 = x + ws[i], inside = t => { const [a, b] = cur(t); return a > x0 - 6 && a < x1 + 6 && b > LY - 22 && b < LY + 14; };
      txt(k, s, x0, LY, F, C.ink);
      // 最後に乗った・離れた時刻から 0.3秒（乗る: 左端から伸びる / 離れる: 左端から縮んで右へ抜ける）
      const on = inside(u); let tc = null; for (let j = 0, t = u; j < 600; j++, t -= 1 / 120) if (inside(t - 1 / 120) !== on) { tc = t; break; }
      if (tc !== null) {
        const q = eOut(P(u, tc, tc + DUR)), a = on ? 0 : q, b = on ? q : 1;
        if (b > a) { k.fillStyle = C.acc; k.fillRect(x0 + ws[i] * a, LY + 8, ws[i] * (b - a), 3); }
      }
      x += ws[i] + GAP;
    });
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
