// 荧光笔标记动画 / Highlighter animation
// 分类: lp (落地页与网页效果) · element 08 · 子类别: 滚动效果
// 示例表现: 文案滚入视口时，黄色荧光笔笔触沿关键词的下半部分从左向右展开。
// 动画参数与实现规则:
//   - 滚动位置在 0.35–1.0 s 从 0→90 px，在 1.6–2.2 s 从 90→180 px，在 3.3–4.4 s 从 180→0，使用三次缓入缓出。
//   - 两行文案使用 20 px 粗体，分别位于页面纵向 190 px 和 290 px 处，最初隐藏在视口下方。
//   - 每行顶边越过触发线后，关键词下方的笔触开始展开；触发线位于视口高度的 80%，即 134 px 处。
//   - 笔触为黄色 #FFE36B，高 11，位于文字下部，从基线上方 8 px 到下方 3 px。宽度在 0.8 s 内从 0 → 关键词宽度，自左向右展开，使用三次缓出，并绘制在文字后方。
//   - 从视口底部离开的文案行会重置为隐藏状态。
// 适用场景: 适合落地页文案，让关键短语在阅读过程中自然突出。
// 标签: highlighter, marker, text-highlight, emphasis, scroll-reveal, copy, underline, 荧光笔, 文字强调, 关键词, 滚动触发, 高亮, 标记动画
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#080
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 08
// 08 マーカーアニメーション（大事な言葉が画面に入ると、蛍光ペンの線が左から引かれる）
CELLS[7] = (k, w, h, u) => {
  frame(k);
  const SY = t => steps([[0.35, 0], [1.0, 90], [1.6, 90], [2.2, 180], [3.3, 180], [4.4, 0]], t), y = SY(u);
  const LINE = VH * 0.8, DUR = 0.8, MARK = '#FFE36B', F = `800 20px ${JP}`;
  const rows = [[190, ['每天的饭菜，', '只需10分钟', '就能做好。']], [290, ['只推荐新手', '也不易失败的食谱', '给你。']]];
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, '为忙碌的你', 70, 50, `700 14px ${JP}`, C.sub); txt(k, '厨房里的新伙伴', 70, 80, `800 24px ${JP}`, C.ink);
    bars(k, 70, 96, 330, 4, C.faint, 6, 12, 0.8);
    for (const [base, parts] of rows) {
      k.font = F; const ws = parts.map(s => k.measureText(s).width); let x = (VW - ws.reduce((a, b) => a + b, 0)) / 2;
      const t0 = enteredAt(SY, base - 20, LINE, u), q = t0 === null ? 0 : eOut(P(u, t0, t0 + DUR));
      if (q > 0) { k.fillStyle = MARK; k.fillRect(x + ws[0], base - 8, ws[1] * q, 11); }
      parts.forEach((s, i) => { txt(k, s, x, base, F, i === 1 ? C.ink : '#4A5368'); x += ws[i]; });
      bars(k, 110, base + 22, 291, 2, C.faint, 6, 12, 0.7);
    }
    k.restore();
    sbar(k, y / 180, 0.5);
  });
};
