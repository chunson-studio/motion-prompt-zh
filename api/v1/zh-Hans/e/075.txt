// 吸顶页头 / Sticky header
// 分类: lp (落地页与网页效果) · element 03 · 子类别: 滚动效果
// 示例表现: 公告条随页面滚出视野，其下方页头到达视口顶边后固定，正文继续从页头下方滚过。
// 动画参数与实现规则:
//   - 滚动位置在 0.4–1.3 s 从 0→110 px，在 1.9–2.8 s 从 110→250 px，在 3.4–4.4 s 从 250→0，使用三次缓入缓出。
//   - 最上方是高 22 px 的深蓝色公告条，下方是高 34 px 的页头，包含标志、三个菜单项和按钮。
//   - 页头位置 = max(22 − scroll, 0)：公告条离开视野后，页头停在顶边，对应 position: sticky; top: 0。
//   - 接下来的 24 px 滚动距离内，页头下方大小为 12 px 的阴影，不透明度从 0 增至 0.16。
//   - 图片、标题和三张卡片等内容在页头下方正常滚动。
// 适用场景: 适合长页面，让导航菜单和行动按钮始终容易找到。
// 标签: sticky, sticky-header, fixed-header, position-sticky, navigation, header, scroll, 吸顶页头, 固定导航, 滚动, 公告条, 阴影, 长页面
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#075
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 03
// 03 追従ヘッダー（スクロールすると、ヘッダーが画面の上で止まってついてくる）
CELLS[2] = (k, w, h, u) => {
  frame(k);
  const y = steps([[0.4, 0], [1.3, 110], [1.9, 110], [2.8, 250], [3.4, 250], [4.4, 0]], u);
  const STRIP = 22;
  view(k, () => {
    k.save(); k.translate(0, -y);
    pic(k, 24, 70, 214, 112, 0, 8);
    txt(k, '开启崭新每一天，', 262, 104, `800 22px ${JP}`, C.ink); bars(k, 262, 120, 210, 3); btn(k, 262, 160, 100, 26, '了解详情');
    k.fillStyle = C.soft; k.fillRect(0, 212, VW, 240);
    txt(k, '特色', VW / 2, 246, `800 18px ${JP}`, C.ink, 'center');
    for (let i = 0; i < 3; i++) { const x = 34 + i * 152; k.fillStyle = '#fff'; rr(k, x, 262, 136, 120, 8); k.fill(); pic(k, x + 8, 270, 120, 58, i, 5); bars(k, x + 10, 340, 110, 3, C.faint, 6, 11); }
    k.fillStyle = C.ink; k.fillRect(0, 0, VW, STRIP);
    txt(k, '限时优惠活动进行中', VW / 2, STRIP / 2 + 0.5, `600 11px ${JP}`, '#fff', 'center', 'middle');
    k.restore();
    const hy = Math.max(STRIP - y, 0), sh = clamp((y - STRIP) / 24);
    if (sh > 0) { const g = k.createLinearGradient(0, hy + 34, 0, hy + 46); g.addColorStop(0, `rgba(28,35,51,${0.16 * sh})`); g.addColorStop(1, 'rgba(28,35,51,0)'); k.fillStyle = g; k.fillRect(0, hy + 34, VW, 12); }
    header(k, hy);
    sbar(k, y / 250, 0.42);
  });
};
