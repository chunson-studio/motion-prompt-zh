// 轻提示 / Toast notification
// 分类: ui (UI 组件) · element 08 · 子类别: 浮层与展开
// 示例表现: 点击“保存”后，深色通知滑入右下角；剩余时间条耗尽后，通知滑出。
// 动画参数与实现规则:
//   - 在 542×191 的卡片上，周期为 4.5 s。光标在 1.0 s 点击左下方的蓝色“保存”按钮，尺寸 110×36，然后向左下方离开。
//   - 通知：颜色 #26282c，尺寸 214×52、圆角半径 10，距右侧 18 px、底部 16 px；在 1.05–1.4 s 从右侧外 240 px 的位置滑入，使用 easeOutCubic（三次缓出）。
//   - 内部：蓝色圆形内有白色对勾，旁边显示 15 px 的白色“已保存”文字。
//   - 底部剩余时间线高 3 px，在 1.4–3.2 s 从长度 194 px 线性缩短至 0。
//   - 退场：在 3.2–3.55 s 向右滑出 240 px，使用 easeInOutCubic（三次缓入缓出）。
// 适用场景: 适合确认保存或发送等结果，同时不中断用户操作。
// 标签: toast, snackbar, notification, alert, message, auto-dismiss, 轻提示, 通知, 保存, 反馈, 自动消失, 消息
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#056
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 08
// 08 トースト（Toast notification）
CELLS[7] = ui((w, h, t) => {
  bar(24, 16, 90, C.line2, 10); strokeRR(24, 34, 300, 34, 7, C.line2, 1.5); bar(24, 82, 90, C.line2, 10); strokeRR(24, 100, 300, 34, 7, C.line2, 1.5);
  fillRR(24, 146, 110, 36, 8, C.blue); text('保存', 79, 164, 16, '#fff', 600, 'center');
  const tin = eo(seg(t, 1.05, 1.4)), tout = eio(seg(t, 3.2, 3.55));
  if (tin > 0 && tout < 1) {
    const tw0 = 214, th = 52, tx = w - 18 - tw0 + (1 - tin) * 240 + tout * 240, ty = h - 16 - th;
    shadowed(() => fillRR(tx, ty, tw0, th, 10, C.dark), 18, 6, .25);
    circle(tx + 26, ty + 24, 11, C.blue); check(tx + 26, ty + 24, 5, '#fff', 2.4);
    text('已保存', tx + 46, ty + 24, 15, '#fff', 600);
    const life = 1 - seg(t, 1.4, 3.2); fillRR(tx + 10, ty + th - 7, (tw0 - 20) * life, 3, 1.5, C.sb);
  }
  pointer([[.3, -30, h + 30], [.95, 84, 168], [1.5, 84, 168], [2.2, -40, h + 40]], [1.0], t);
});
