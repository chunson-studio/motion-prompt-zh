// 预加载画面 / Preloader
// 分类: lp (落地页与网页效果) · element 17 · 子类别: 加载、导航与循环动效
// 示例表现: 深蓝色屏幕中，标志下方的数字从 0% 递增到 100%；随后整屏向上移出，下方页面淡入。
// 动画参数与实现规则:
//   - 深蓝色屏幕铺满视口，内含标志、字号 28 px 的数字和尺寸 180 × 4 的进度条。
//   - 进度在 0.25–2.1 s 从 0→100%，使用三次缓出并取整；进度条以相同比例填充为橙色。
//   - 在 2.3–2.85 s，整屏向上移动 168 px 并移出，使用三次缓入缓出。
//   - 页面标题和图片在 2.35–2.95 s 内，不透明度从 0→1，并从下方 14 px 处上移归位，使用三次缓出。
//   - 在 4.25–4.75 s，预加载屏从顶部落回，进度重置为 0%，开始下一次加载。
// 适用场景: 适合图片或视频较多的网站，将加载等待设计为品牌展示的一部分。
// 标签: preloader, loading-screen, loader, percentage, intro, splash, curtain, 预加载, 加载进度, 开场, 品牌, 进度条, 百分比
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#089
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 17
// 17 ローディング画面（数字が100%まで進むと、幕が上がってページが現れる）
CELLS[16] = (k, w, h, u) => {
  frame(k);
  const n = Math.round(100 * eOut(P(u, 0.25, 2.1)));
  const cy = u < 4.25 ? -VH * eIO(P(u, 2.3, 2.85)) : -VH * (1 - eIO(P(u, 4.25, 4.75)));
  const show = eOut(P(u, 2.35, 2.95));
  view(k, () => {
    header(k, 0);
    k.save(); k.globalAlpha = show; k.translate(0, 14 * (1 - show));
    txt(k, '欢迎光临。', 34, 82, `800 26px ${JP}`, C.ink); bars(k, 34, 98, 200, 3); btn(k, 34, 136, 96, 24, '开始使用');
    pic(k, 270, 46, 214, 112, 0, 8); k.restore();
    if (cy > -VH) {
      k.save(); k.translate(0, cy); k.fillStyle = C.ink; k.fillRect(0, 0, VW, VH);
      logo(k, VW / 2 - 40, 56, '#fff');
      txt(k, (u >= 4.25 ? 0 : n) + '%', VW / 2, 98, `800 28px ${EN}`, '#fff', 'center', 'middle');
      k.fillStyle = 'rgba(255,255,255,.15)'; rr(k, VW / 2 - 90, 122, 180, 4, 2); k.fill();
      k.fillStyle = C.acc; rr(k, VW / 2 - 90, 122, 180 * (u >= 4.25 ? 0 : n / 100), 4, 2); k.fill();
      k.restore();
    }
  });
};
