// 横向滚动 / Horizontal scroll
// 分类: lp (落地页与网页效果) · element 05 · 子类别: 滚动效果
// 示例表现: 页面向下滚动时，展示区块保持不动，四张带编号的作品卡片向左移动。
// 动画参数与实现规则:
//   - 滚动位置在 0.3–3.0 s 从 0→420 px，在 3.5–4.5 s 从 420→0，使用三次缓入缓出。
//   - 标题“WORKS”先滚出视野；滚动达到 40 px 时，区块到达顶边并固定。
//   - 四张卡片的尺寸为 200 × 112，间距 20，两侧留白 34。水平移动总距离 = 34·2 + 200·4 + 20·3 − 511 = 417 px。
//   - 卡片 x = 起始 x − p × 417，其中 p = (scroll − 40) / 380；位置与滚动距离线性对应，不使用缓动。
//   - 卡片横向移动时，滚动条滑块仍向下移动。
// 适用场景: 适合在纵向长页面中，用横排形式展示作品、案例或产品。
// 标签: horizontal-scroll, sideways, scrolltrigger, pin, gallery, portfolio, scroll, 横向滚动, 作品集, 卡片, 横排, 滚动固定, 案例展示
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#077
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 05
// 05 横スクロール（縦にスクロールすると、カードの列が横へ流れる）
CELLS[4] = (k, w, h, u) => {
  frame(k);
  const y = steps([[0.3, 0], [3.0, 420], [3.5, 420], [4.5, 0]], u);
  const PIN0 = 40, PD = 380, N = 4, PWD = 200, GAP = 20;
  const top = y < PIN0 ? PIN0 - y : 0, p = clamp((y - PIN0) / PD);
  const travel = 34 * 2 + N * PWD + (N - 1) * GAP - VW;
  view(k, () => {
    txt(k, 'WORKS', 34, 30 - y, `800 20px ${EN}`, C.ink, 'left', 'alphabetic', '2px');
    k.save(); k.translate(0, top);
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH + 40);
    for (let i = 0; i < N; i++) {
      const x = 34 + i * (PWD + GAP) - p * travel;
      pic(k, x, 18, PWD, 112, i % 3, 8);
      txt(k, '0' + (i + 1), x + 14, 118, `800 22px ${EN}`, '#fff');
      bars(k, x, 140, 120, 1, '#CBD2DE');
    }
    k.restore();
    sbar(k, y / 420, 0.36);
  });
};
