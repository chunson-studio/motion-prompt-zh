// 黄金时刻 / Golden hour
// 分类: photo (摄影效果、光线与风格) · element 07 · 子类别: 布光与时段
// 示例表现: 太阳下降到地平线稍上方，天空与地面转为橙色，人物左侧被照亮，路灯与树木的影子向右拉长。
// 动画参数与实现规则:
//   - 场景 1：人物半身像位于 y 150 的地平线前方，头部圆心为 x 270、y 78，半径 29 px，不绘制面部或头发。
//   - 太阳半径 18 px，从 (x 110, y −30) 移到 (x 96, y 136)，并被山丘遮住一半。
//   - 颜色：天空从 #FAF9F7 变为顶部的 #FFE8BC 和地平线附近的 #FFCF80；地面从 #F1F0EE → #F6DDB0；人物亮部为 #FFB547，阴影为 #3E3650。
//   - 头部光线以约 66° 从左侧照来，分界右侧为阴影；躯干左侧受光，右侧处于阴影。
//   - 影子从路灯与树木底部向右延伸，长度从 0 → 150 px，颜色为 #DDAA6E；右上角时间从 12:00 → 17:40。
//   - 时序：效果在 0.4 s 至 1.4 s 进入，保持到 4.0 s，并在 4.8 s 前完全消失；单次周期 = 5 s，使用三次缓入缓出，约等于 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合温暖、怀旧、柔和的画面氛围。
// 标签: golden hour, magic hour, sunset light, warm light, long shadows, low sun, 黄金时刻, 日落, 暖色, 长阴影, 夕阳, 氛围
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#103
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 07
// ゴールデンアワー: 太陽が地平線近くまで下り、全体が橙に染まって影が長く伸びる
CELLS[6] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  const GOLD = {
    sky: '#FFE8BC', sky2: '#FFCF80', far: '#F2C58C', mid: '#E8B27A', ground: '#F6DDB0', ground2: '#ECC892', mark: '#F6DDB0', cast: '#DDAA6E',
    bld: '#F7DCAE', edge: '#D9AE76', win: '#ECC892', winOff: '#ECC892', tree: '#E9C08A', trunk: '#C9965E', post: '#B98A58', bulb: '#C9965E',
    person: '#5C5470', shade: '#3E3650', lit: '#FFB547', car: '#A08E8E', glass: '#F6DDB0', tire: '#3E3650', sun: '#F59A24', glow: '#FFD48A'
  };
  const p = pal(BASE, GOLD, a);
  stagePortrait(k, p, { sun: [lerp(110, 96, a), lerp(-30, 136, a), 18, 1], cast: lerp(0, 150, a) });
  const lit = mixc(p.person, GOLD.lit, a), sh = mixc(p.person, GOLD.shade, a);
  bust(k, PX, PY, R, p.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
    headShade(kk, PX, PY, R, 1.15, 1, sh);
    torso(kk, PX, PY, R, PX - W, PX - 0.5 * W, lit);
    torso(kk, PX, PY, R, PX + 0.4 * W, PX + W, sh);
  });
  hud(k, clock(lerp(720, 1060, a)));
});
