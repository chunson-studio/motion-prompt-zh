// 页面转场 / Page transition
// 分类: lp (落地页与网页效果) · element 20 · 子类别: 加载、导航与循环动效
// 示例表现: 点击作品列表中的卡片，橙色面板从左侧覆盖页面；遮挡期间切换网址和详情页，面板再向右移出。“返回列表”使用同样的过渡。
// 动画参数与实现规则:
//   - 在 0.8 s 点击卡片。在 0.9–1.3 s，橙色面板从左侧进入并覆盖整个视口，中央显示标志，使用三次缓入缓出。
//   - 在 1.3 s，画面完全被遮挡时，页面与地址栏网址同时切换：/works → /works/02。
//   - 在 1.4–1.85 s，面板向右移出，使用三次缓入缓出。
//   - 在 3.0 s 点击“← 返回列表”：面板在 3.1–3.5 s 进入，页面在 3.5 s 切换，面板在 3.6–4.05 s 移出。
//   - 面板覆盖整个视口，包括页头。
// 适用场景: 适合品牌网站或作品集，让相邻页面之间的切换形成连贯过渡。
// 标签: page-transition, transition, view-transitions, navigation, wipe, barba, spa, 页面转场, 遮罩, 页面切换, 作品集, 覆盖, 过渡动画
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#092
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 20
// 20 ページ遷移アニメーション（リンクを押すと、色の幕が横切って次のページに切り替わる）
CELLS[19] = (k, w, h, u) => {
  frame(k, u >= 1.3 && u < 3.5 ? 'example.com/works/02' : 'example.com/works');
  const CLK = [0.8, 3.0];
  const cover = t0 => eIO(P(u, t0, t0 + 0.4)), uncover = t0 => eIO(P(u, t0, t0 + 0.45));
  // 幕の左端: 入る −VW→0、出る 0→VW
  const xs = u < 1.4 ? -VW * (1 - cover(0.9)) : u < 3.1 ? VW * uncover(1.4) : u < 3.6 ? -VW * (1 - cover(3.1)) : VW * uncover(3.6);
  const pageB = u >= 1.3 && u < 3.5;
  const [cx, cy] = path([[0, 420, 150], [0.65, 255, 100], [2.6, 255, 100], [2.85, 46, 50], [3.3, 46, 50], [4.6, 420, 150], [4.8, 420, 150]], u);
  view(k, () => {
    header(k, 0);
    if (!pageB) {
      txt(k, 'WORKS', 34, 62, `800 18px ${EN}`, C.ink, 'left', 'alphabetic', '2px');
      for (let i = 0; i < 3; i++) { const x = 34 + i * 152; pic(k, x, 74, 136, 60, i, 6); bars(k, x, 142, 100, 1, C.ink, 7); }
    } else {
      txt(k, '← 返回列表', 34, 55, `700 12px ${JP}`, C.sub, 'left', 'middle');
      pic(k, 34, 68, 240, 90, 1, 8);
      txt(k, 'Project 02', 296, 96, `800 22px ${EN}`, C.ink); bars(k, 296, 110, 180, 3);
    }
    if (xs > -VW && xs < VW) { k.fillStyle = C.acc; k.fillRect(xs, 0, VW, VH); logo(k, xs + VW / 2 - 40, VH / 2, '#fff'); }
    click(k, cx, cy, u, CLK);
  });
};
