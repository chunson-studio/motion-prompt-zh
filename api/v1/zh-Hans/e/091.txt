// 无限滚动 / Infinite scroll
// 分类: lp (落地页与网页效果) · element 19 · 子类别: 加载、导航与循环动效
// 示例表现: 向下滚动到列表末尾时出现加载转圈，随后追加三条内容，让阅读连续进行。
// 动画参数与实现规则:
//   - 列表项尺寸为 331 × 44，圆角半径 8，纵向间隔 52；最后一项下方显示半径 10 的加载转圈，每 0.8 s 旋转一周。
//   - 滚动位置在 0.25–0.85 s 从 38→116 px，在 1.15–1.75 s 从 116→194 px，使用三次缓入缓出；列表末尾进入视口后停止滚动。
//   - 在 2.6 s 追加三条内容：每条在 0.35 s 内将不透明度从 0→1，并从下方 10 px 处上移，使用三次缓出。加载转圈移动到新的列表末尾。
//   - 每个周期向前推进三条内容，即 156 px。内容每三条重复，因此周期末尾与开头的画面一致。
// 适用场景: 适合信息流或新闻列表，让用户无需翻页即可持续阅读。
// 标签: infinite-scroll, endless-scroll, load-more, feed, pagination, spinner, list, 无限滚动, 信息流, 追加内容, 列表, 加载更多, 连续阅读
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#091
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 19
// 19 無限スクロール（いちばん下まで来ると読み込み表示が出て、続きが足される）
CELLS[18] = (k, w, h, u) => {
  frame(k);
  // 1周期で 3件（156px）ぶん進む。周期の終わりは始まりと同じ見た目（内容は3件ごとにくり返す）
  const PITCH = 52, BATCH = 3, L0 = 6, TL = 2.6;
  const y = steps([[0.25, 38], [0.85, 116], [1.15, 116], [1.75, 194], [4.8, 194]], u);
  const n = u < TL ? L0 : L0 + BATCH, end = 10 + n * PITCH;
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    k.save(); k.translate(0, -y);
    for (let i = 0; i < n; i++) {
      const top = 10 + i * PITCH, v = i % 3, fresh = i >= L0 ? eOut(P(u, TL, TL + 0.35)) : 1;
      if (top - y > VH || top + 44 - y < 0) continue;
      k.save(); k.globalAlpha = fresh; k.translate(0, 10 * (1 - fresh));
      k.fillStyle = '#fff'; rr(k, 90, top, 331, 44, 8); k.fill(); pic(k, 98, top + 6, 48, 32, v, 5);
      bars(k, 158, top + 12, [180, 140, 200][v], 1, C.ink, 7); bars(k, 158, top + 26, [120, 150, 100][v], 1, C.faint, 6);
      k.restore();
    }
    spinner(k, VW / 2, end + 18, 10, u);
    k.restore();
  });
};
