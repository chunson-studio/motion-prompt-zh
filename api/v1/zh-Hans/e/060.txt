// 分段控制器 / Segmented control
// 分类: ui (UI 组件) · element 12 · 子类别: 切换与输入
// 示例表现: 点击“日、周、月”，蓝色选中背景横向滑动，下方柱状图的高度随之变化。
// 动画参数与实现规则:
//   - 在 542×191 的卡片上，周期为 4.5 s。位置 (121, 12) 有一个 300×40 的灰色底座，圆角半径 10；划分为三个宽 100 的区段，分别为日、周、月。
//   - 选中背景为蓝色，尺寸 92×32、圆角半径 8，阴影模糊半径 6；选中标签为白色。
//   - 光标在 1.0 s 点击周，在 2.1 s 点击月，在 3.3 s 点击日；每次选中背景在 0.25 s 内滑动，使用 easeInOutCubic（三次缓入缓出）。
//   - 下方图表有七根宽 34 的柱条，间隔 58 px，第六根为蓝色；高度随选中项在三组数据之间变化，范围约为 30–85、40–100、60–115，使用相同缓动。
// 适用场景: 适合在同一内容的两至四个视图之间切换，例如日、周、月。
// 标签: segmented-control, segmented-button, button-group, toggle-group, switcher, view-switch, 分段控制器, 切换, 日期, 视图, 图表, 选中
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#060
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 12
// 12 セグメントコントロール（Segmented control）
CELLS[11] = ui((w, h, t) => {
  const pos = eio(seg(t, 1.0, 1.25)) + eio(seg(t, 2.1, 2.35)) - 2 * eio(seg(t, 3.3, 3.55));
  const sx = 121, sy = 12, sw = 300, sh = 40, cw = sw / 3;
  fillRR(sx, sy, sw, sh, 10, '#efedeb');
  shadowed(() => fillRR(sx + 4 + pos * cw, sy + 4, cw - 8, sh - 8, 8, C.blue), 6, 2, .2);
  ['日', '周', '月'].forEach((s, i) => { const near = 1 - clamp(Math.abs(pos - i)); text(s, sx + cw * i + cw / 2, sy + sh / 2 + 1, 17, near > .5 ? '#fff' : '#6d6b69', 700, 'center'); });
  const sets = [[30, 55, 40, 70, 50, 85, 60], [60, 40, 90, 75, 100, 55, 80], [95, 70, 110, 60, 85, 115, 100]];
  const fi = Math.floor(pos), fr = pos - fi, gi = Math.min(2, fi + 1);
  X.fillStyle = C.line; X.fillRect(60, 182, 422, 1.5);
  for (let k = 0; k < 7; k++) { const bh = lerp(sets[fi][k], sets[gi][k], eio(fr)); fillRR(72 + k * 58, 180 - bh, 34, bh, 5, k === 5 ? C.blue : C.mb); }
  pointer([[.3, w + 30, h + 30], [.95, 276, 36], [1.6, 276, 36], [2.05, 376, 36], [2.8, 376, 36], [3.25, 176, 36], [3.7, 176, 36], [4.25, w + 30, h + 30]], [1.0, 2.1, 3.3], t);
});
