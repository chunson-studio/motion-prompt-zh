// 滚动淡入 / Fade-in on scroll
// 分类: lp (落地页与网页效果) · element 02 · 子类别: 滚动效果
// 示例表现: 随着页面滚动，标题、卡片和正文进入可视区域时分别上移归位并淡入。
// 动画参数与实现规则:
//   - 滚动位置在 0.35–1.0 s 从 0→95 px，在 1.5–2.1 s 从 95→200 px，在 3.2–4.4 s 从 200→0；均使用三次缓入缓出。
//   - 触发线位于可视区域高度的 85%，即距顶部 143 px。元素顶边越过这条线时触发，体现 Intersection Observer 的工作思路。
//   - 触发后在 0.7 s 内，不透明度从 0→1，向下偏移从 28 px → 0，使用三次缓出。
//   - 三个元素分别是页面纵向 172 px 处的标题、206 px 处由三张卡片组成的一行，以及 334 px 处的正文；最初都在视口下方。同一行卡片一起出现。
//   - 页面向上返回时，元素一旦从视口底部离开，就重置为隐藏状态，下次进入时再次播放。
// 适用场景: 适合长落地页，在读者滚动时逐段呈现内容。
// 标签: fade-in, scroll-reveal, reveal-on-scroll, intersection-observer, aos, entrance, scroll, 滚动淡入, 入场, 上移, 落地页, 逐段呈现, 视口触发
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#074
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 02
// 02 スクロールでフェードイン（画面に入った要素が、下から浮かび上がって現れる）
CELLS[1] = (k, w, h, u) => {
  frame(k);
  const SY = t => steps([[0.35, 0], [1.0, 95], [1.5, 95], [2.1, 200], [3.2, 200], [4.4, 0]], t), y = SY(u);
  const LINE = VH * 0.85, DUR = 0.7, RISE = 28;
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, '让生活，轻一点。', 34, 52, `800 24px ${JP}`, C.ink); bars(k, 34, 68, 190, 3); btn(k, 34, 108, 96, 26, '了解详情');
    pic(k, 286, 18, 196, 110, 0, 8);
    const items = [
      [172, () => txt(k, '选择我们的3个理由', VW / 2, 172 + 20, `800 18px ${JP}`, C.ink, 'center')],
      [206, () => { for (let i = 0; i < 3; i++) { const x = 34 + i * 152; k.fillStyle = C.soft; rr(k, x, 206, 136, 106, 8); k.fill(); pic(k, x + 8, 214, 120, 56, i, 5); bars(k, x + 10, 280, 112, 2, '#CBD2DE'); } }],
      [334, () => bars(k, 110, 334, 290, 3, C.faint, 7, 13)]
    ];
    for (const [top, draw] of items) {
      const t0 = enteredAt(SY, top, LINE, u); if (t0 === null) continue;
      const q = eOut(P(u, t0, t0 + DUR));
      k.save(); k.globalAlpha = q; k.translate(0, RISE * (1 - q)); draw(); k.restore();
    }
    k.restore();
    sbar(k, y / 200, 0.5);
  });
};
