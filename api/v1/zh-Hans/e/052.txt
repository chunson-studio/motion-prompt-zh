// 折叠面板 / Accordion
// 分类: ui (UI 组件) · element 04 · 子类别: 浮层与展开
// 示例表现: 点击三个标题中的第一项，其内容展开，并将下面各行向下推；再次点击则收起。
// 动画参数与实现规则:
//   - 在 542×191 的卡片上，周期为 4.5 s。三个标题行宽 494、高 34，圆角半径 8、间隔 6，右侧各有一个向下箭头。
//   - 光标在 0.95 s 到达第一个标题，并在 1.0 s 和 3.3 s 点击。
//   - 展开进度 ex 在 1.0–1.35 s 从 0→1，使用 easeInOutCubic（三次缓入缓出），再在 3.3–3.62 s 从 1→0。
//   - 第一行高度从 34 增至 92，增量为 +58·ex，并将下方行向下推；内部两行内容裁切在该行范围内，并以不透明度 ex 渐入。
//   - 展开时，箭头旋转 180°·ex，边框变为浅蓝色，标题变为蓝色。
// 适用场景: 适合常见问题或较长的详情，让用户先浏览标题，只展开需要的部分。
// 标签: accordion, collapsible, expand, collapse, disclosure, faq, details, 折叠面板, 手风琴, 展开, 收起, 常见问题, 详情
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#052
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 04
// 04 アコーディオン（Accordion）
CELLS[3] = ui((w, h, t) => {
  const ex = eio(seg(t, 1.0, 1.35)) * (1 - eio(seg(t, 3.3, 3.62)));
  let y = 12; const H0 = 34, EX = 58;
  for (let i = 0; i < 3; i++) {
    const extra = i === 0 ? EX * ex : 0;
    strokeRR(24, y, 494, H0 + extra, 8, i === 0 && ex > .05 ? C.sb : C.line2, 1.5);
    bar(42, y + 12, [180, 220, 150][i], i === 0 && ex > .05 ? C.blue : C.line2, 10);
    chevron(494, y + 17, 6, i === 0 ? Math.PI * ex : 0, i === 0 && ex > .05 ? C.blue : '#9a9896');
    if (i === 0 && ex > 0) { X.save(); rr(24, y, 494, H0 + extra, 8); X.clip(); X.globalAlpha = ex; bar(42, y + H0 + 6, 420); bar(42, y + H0 + 24, 320); X.restore(); }
    y += H0 + extra + 6;
  }
  pointer([[.3, w + 30, h + 30], [.95, 300, 31], [1.6, 300, 31], [3.0, 300, 31], [3.55, 300, 31], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
