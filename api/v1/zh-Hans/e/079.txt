// 滚动进度条 / Scroll progress bar
// 分类: lp (落地页与网页效果) · element 07 · 子类别: 滚动效果
// 示例表现: 文章分段向下滚动时，顶部进度线从左向右延伸；到达页面底部时，进度线填满。
// 动画参数与实现规则:
//   - 文章从 0→430 px 滚动，分为四次、每次 110 px 的步进；每次移动 0.45 s 后停顿 0.4 s，使用三次缓入缓出。
//   - 在 3.75–4.55 s 从 430→0 返回。
//   - 沿视口顶部绘制高 6 px 的进度条，轨道浅灰，填充橙色。
//   - 填充宽度 = 511 px × scroll / 430，直接使用滚动比例，与 CSS animation-timeline: scroll() 的含义相同。
// 适用场景: 适合长文章，提示读者还剩多少内容未读。
// 标签: progress-bar, reading-progress, scroll-progress, indicator, scroll-driven, article, 滚动进度条, 阅读进度, 长文章, 顶部, 进度, 网页
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#079
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 07
// 07 スクロールプログレスバー（読み進めた分だけ、画面上の線が伸びる）
CELLS[6] = (k, w, h, u) => {
  frame(k);
  const MAX = 430;
  const y = steps([[0.3, 0], [0.75, 110], [1.15, 110], [1.6, 220], [2.0, 220], [2.45, 330], [2.85, 330], [3.3, MAX], [3.75, MAX], [4.55, 0]], u);
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, 'Web设计入门', 70, 44, `800 20px ${JP}`, C.ink); bars(k, 70, 56, 120, 1, C.faint, 5);
    pic(k, 70, 74, 371, 112, 0, 8);
    for (let b = 0; b < 4; b++) { const y0 = 204 + b * 82; k.fillStyle = C.ink; rr(k, 70, y0, 140, 9, 4.5); k.fill(); bars(k, 70, y0 + 22, 371, 4, C.faint, 6, 12); }
    k.restore();
    k.fillStyle = C.faint; k.fillRect(0, 0, VW, 6);
    k.fillStyle = C.acc; k.fillRect(0, 0, VW * (y / MAX), 6);
    sbar(k, y / MAX, 0.3);
  });
};
