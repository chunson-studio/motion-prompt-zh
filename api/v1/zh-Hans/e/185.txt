// 插入镜头 / Insert shot
// 分类: camera2 (镜头运动与转场（第 2 弹）) · element 17 · 子类别: 转场与剪辑
// 示例表现: 在场景中插入人物腕表的大特写，持续 1.3 s，然后返回原镜头。
// 动画参数与实现规则:
//   - 衔接：两次直接剪切，过渡为 0 帧；在 1.6 s 切到细节，在 2.9 s 返回原场景，细节镜头持续 1.3 s。
//   - 主镜头 A：相机固定，高度 1.4 m；人物位于左侧 0.6 m、前方 6 m，腕部有一块手表，表现为深色圆点；一根立柱位于右侧 2.2 m、前方 4 m。
//   - 细节 INSERT：来自同一场景的手腕大特写；浅蓝色背景上是一段蓝色袖子，尺寸 600×92 px、旋转 -0.35 rad；白色表盘半径 58 px，带深色指针，蓝色秒针在 1.3 s 内转一周。
//   - 底部时间条：分为灰色 A、蓝色细节、灰色 A 三段，并带播放头；插入的内容属于同一场景，本组第 21 项切出镜头则展示另一个主体。
//   - 右上角读数依次为“SHOT A”→“INSERT”→“SHOT A”；在 4.7 s 直接切回起始状态。
// 适用场景: 适合展示小物件或手部动作，或强调重要道具。
// 标签: insert shot, insert, detail, close-up, cut in, prop, edit, 镜头, 插入, 细节, 道具, 特写, 剪辑
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit camera2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera2.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#185
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 17
// 17 インサートショット: 同じ場面の細部（手元の時計）を1.3秒だけ差し込み、元の画に戻る
const INS = [1.6, 2.9];
CELLS[16] = shot((k, w, h, u) => {
  const ins = u >= INS[0] && u < INS[1];
  if (!ins) world(k, w, h, cam({ y: 1.4 }), { person: { x: -0.6, z: 6, walk: 0 }, fg: [[2.2, 4]], after: (kk, cc) => { const q = proj(cc, -0.35, 1.08, 6); kk.fillStyle = C.ink; kk.beginPath(); kk.arc(q.x, q.y, 1.6 * q.s / 10, 0, 7); kk.fill(); } });
  else {   // 手首の時計の大写し
    k.save(); k.translate(w / 2, h / 2);
    k.fillStyle = C.blueXL; k.fillRect(-w, -h, w * 2, h * 2);
    k.fillStyle = C.blue; k.save(); k.rotate(-0.35); rr(k, -300, -46, 600, 92, 30); k.fill(); k.restore();
    k.fillStyle = '#fff'; k.beginPath(); k.arc(0, 0, 58, 0, 7); k.fill(); k.strokeStyle = C.ink; k.lineWidth = 5; k.stroke();
    k.strokeStyle = C.ink; k.lineWidth = 4; k.lineCap = 'round'; k.beginPath(); k.moveTo(0, 0); k.lineTo(0, -34); k.moveTo(0, 0); k.lineTo(26, 14); k.stroke();
    const a = (u - INS[0]) / (INS[1] - INS[0]) * Math.PI * 2 - Math.PI / 2;   // 秒針が1周
    k.strokeStyle = C.blue; k.lineWidth = 2; k.beginPath(); k.moveTo(0, 0); k.lineTo(Math.cos(a) * 44, Math.sin(a) * 44); k.stroke();
    k.restore();
  }
  strip(k, w, h, [[0, INS[0], 'A', C.g2], [INS[0], INS[1], '细节', C.blue], [INS[1], CYC, 'A', C.g2]], u);
  return ins ? 'INSERT' : 'SHOT A';
});
