// 色差 / Chromatic aberration
// 分类: photo (摄影效果、光线与风格) · element 15 · 子类别: 镜头与相机效果
// 示例表现: 街景分离为红、绿、蓝三层，再以略微不同的大小重新叠加；人物、建筑和路灯边缘出现红色与青色分边，越靠近画面两侧越明显。
// 动画参数与实现规则:
//   - 场景 2：街角中，建筑、树木和路灯（x 114、292、494）位于 y 132–162 的道路后方；人物站在左前方，脚底位于 x 176、y 200，身高 124 px。
//   - 分离：图像分别与红色 #FF0000、绿色 #00FF00、蓝色 #0000FF 相乘，得到三个副本，再在黑色背景上加色合成。
//   - 围绕画面中心缩放：红色层为 1 + 2s，绿色层为 1 + s，蓝色层为 1，其中 s 从 0 → 0.02。
//   - 偏移：红蓝两层在左右边缘最终相距约 11 px，在中心相距 0。
//   - 时序：效果在 0.4 s 至 1.4 s 进入，保持到 4.0 s，并在 4.8 s 前完全消失；单次周期 = 5 s，使用三次缓入缓出，约等于 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合表现旧镜头、VHS 或故障画面的不完美质感。
// 标签: chromatic aberration, color fringing, rgb split, lens aberration, fringe, glitch, 色差, 色散, 红蓝分离, 旧镜头, 故障, 边缘偏移
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#111
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 15
// 色収差: 赤・緑・青を少しずつ違う大きさで重ね、画面の端ほど縁に赤と青のずれが出る
CELLS[14] = cell((k, u) => {
  const a = env(u), s = 0.02 * a, cx = CW / 2, cy = CH / 2;
  if (s < 0.0005) { stageStreet(k, BASE); return; }
  const [S, ks] = layer('ca'); stageStreet(ks, BASE);
  k.fillStyle = '#000000'; k.fillRect(0, 0, CW, CH);
  for (const [col, sc] of [['#FF0000', 1 + 2 * s], ['#00FF00', 1 + s], ['#0000FF', 1]]) {
    const [L, kl] = layer('ca-' + col); kl.drawImage(S, 0, 0); kl.globalCompositeOperation = 'multiply'; kl.fillStyle = col; kl.fillRect(0, 0, CW, CH); kl.globalCompositeOperation = 'source-over';
    k.save(); k.globalCompositeOperation = 'lighter'; k.translate(cx, cy); k.scale(sc, sc); k.translate(-cx, -cy); k.drawImage(L, 0, 0); k.restore();
  }
});
