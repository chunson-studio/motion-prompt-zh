// 滚动隐藏页头 / Hide-on-scroll header
// 分类: lp2 (落地页与网页效果（第 2 弹）) · element 05 · 子类别: 滚动效果
// 示例表现: 向下滚动时，页头向上滑出以腾出屏幕空间；只需稍微向上滚动，页头就会立即出现。
// 动画参数与实现规则:
//   - 视口为 511 × 168，顶部固定一个高 34 px 的页头。滚动距离在 0.3–1.1 s 从 0→170 px，向下；在 1.5–1.9 s 从 170→100，向上；在 2.4–2.9 s 从 100→230，向下；在 3.3–4.3 s 从 230→0，向上。每段均使用三次缓入缓出。一次往返为 4.7 s，每个 9.4 s 的效果包周期内执行两次。
//   - 最近一次滚动方向向下时隐藏页头，向上时显示页头。滚动位置距离页面顶部不足 34 px 时，页头始终显示。
//   - 隐藏与显示均在原位与向上偏移 36 px 的位置之间切换，持续 0.3 s，使用二次缓入缓出，类似带 transition 的 transform: translateY。
//   - 页面滚动后，页头下方出现模糊半径 12 px 的阴影；前 30 px 滚动距离内，阴影不透明度从 0→0.14。滚动条滑块随滚动位置变化。
// 适用场景: 适合手机等屏幕空间有限的界面，为内容留出更多空间，同时让用户稍微回滚即可找到菜单。
// 标签: header, hide-on-scroll, headroom, navigation, sticky-header, scroll-direction, 滚动, 页头, 导航, 隐藏, 移动端, 屏幕空间
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit lp2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp2.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#149
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 05
// 05 隠れるヘッダー（下へスクロールするとヘッダーが上へ引っ込み、上へ戻すとすっと出てくる）
CELLS[4] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const SY = t => steps([[0.3, 0], [1.1, 170], [1.5, 170], [1.9, 100], [2.4, 100], [2.9, 230], [3.3, 230], [4.3, 0]], t), y = SY(u);
  // 最後に動いた向きが「下」なら隠す（ページの一番上にいるときは出す）
  const hidden = t => { const yt = SY(t); if (yt < 34) return false; for (let i = 0, tt = t; i < 600; i++, tt -= 1 / 120) { const d = SY(tt) - SY(tt - 1 / 120); if (Math.abs(d) > 1e-4) return d > 0; } return false; };
  const q = trans(hidden, u, 0.3, eIOs);
  view(k, () => {
    k.save(); k.translate(0, -y);
    pic(k, 24, 48, 214, 110, 1, 8);
    txt(k, '旅行准备，', 262, 84, `800 22px ${JP}`, C.ink); txt(k, '一站搞定。', 262, 114, `800 22px ${JP}`, C.ink); bars(k, 262, 128, 200, 2); btn(k, 262, 152, 100, 24, '了解详情');
    k.fillStyle = C.soft; k.fillRect(0, 196, VW, 260);
    txt(k, '热门套餐', VW / 2, 232, `800 17px ${JP}`, C.ink, 'center');
    for (let r = 0; r < 2; r++) for (let i = 0; i < 3; i++) { const x = 34 + i * 152, y0 = 250 + r * 100; k.fillStyle = '#fff'; rr(k, x, y0, 136, 88, 8); k.fill(); pic(k, x + 8, y0 + 8, 120, 46, (i + r) % 3, 5); bars(k, x + 10, y0 + 64, 100, 2, C.faint, 5, 10); }
    k.restore();
    const hy = -36 * q;
    if (y > 0 && q < 1) { const g = k.createLinearGradient(0, hy + 34, 0, hy + 46); g.addColorStop(0, `rgba(28,35,51,${0.14 * clamp(y / 30)})`); g.addColorStop(1, 'rgba(28,35,51,0)'); k.fillStyle = g; k.fillRect(0, hy + 34, VW, 12); }
    header(k, hy);
    sbar(k, y / 230, 0.42);
  });
};
