// 蝴蝶光 / Butterfly lighting
// 分类: photo (摄影效果、光线与风格) · element 06 · 子类别: 布光与时段
// 示例表现: 位于脸部正前方高处的光源亮起，面部两侧均匀受光，下巴下方只留一小块对称阴影。
// 动画参数与实现规则:
//   - 场景 1：人物半身像位于 y 150 的地平线前方，头部圆心为 x 270、y 78，半径 29 px，不绘制面部或头发。背景变暗，天空颜色为 #E7E5E1。
//   - 头部全部为 #FFC75A 亮部，没有阴影侧。
//   - 下巴下方的阴影为椭圆，中心位于 y 78+1.06R，宽 0.72R、高 0.38R；R = 29，约 21×11 px，颜色为 #33405A。
//   - 躯干中心 ±0.8×56 px 范围内为亮部，外缘窄条保留底色；布光图显示光源位于镜头一侧的 0°，右上角读数为“0° · ↑45°”。
//   - 时序：效果在 0.4 s 至 1.4 s 进入，保持到 4.0 s，并在 4.8 s 前完全消失；单次周期 = 5 s，使用三次缓入缓出，约等于 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合希望面部光线柔美、讨喜的美容或时尚人像。
// 标签: butterfly lighting, paramount lighting, beauty lighting, portrait lighting, overhead light, glamour, 蝴蝶光, 正面光, 高位光, 美容, 时尚人像, 布光
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 原版动态示例: https://showheyohtaki.github.io/motion-prompt/#102
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用前已清空画布并由 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为单个效果的代码。CELLS、常量和辅助函数在对应的 kit 文件中。原始代码与版权声明按 MIT 许可保留。

// @element 06
// バタフライライティング: 顔の正面・高い位置からの光。顔は左右対称に明るく、あごの下にだけ小さな影
CELLS[5] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a);
  bust(k, PX, PY, R, BASE.person, kk => {
    torso(kk, PX, PY, R, PX - 0.8 * W, PX + 0.8 * W, lit);
    kk.fillStyle = sh; kk.beginPath(); kk.ellipse(PX, PY + R * 1.06, R * 0.72, R * 0.38, 0, 0, TAU); kk.fill();    // あごの下の影（左右対称）
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
  });
  plan(k, [[0, a]]);
  hud(k, '0° · ↑45°');
});
